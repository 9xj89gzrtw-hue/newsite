    15→### 1. Sticky Header with Scroll Effect (M Culinary style)
    16→
    17→```tsx
    18→// components/catering/site-header.tsx (enhanced)
    19→'use client';
    20→
    21→import { useState, useEffect } from 'react';
    22→import { motion, useScroll, useTransform } from 'framer-motion';
    23→import Link from 'next/link';
    24→
    25→export function SiteHeader() {
    26→  const [isScrolled, setIsScrolled] = useState(false);
    27→  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);
    28→  
    29→  const { scrollY } = useScroll();
    30→  const headerBg = useTransform(
    31→    scrollY,
    32→    [0, 100],
    33→    ['rgba(26, 22, 20, 0)', 'rgba(26, 22, 20, 0.95)']
    34→  );
    35→  const headerBlur = useTransform(
    36→    scrollY,
    37→    [0, 100],
    38→    ['blur(0px)', 'blur(20px)']
    39→  );
    40→
    41→  useEffect(() => {
    42→    const handleScroll = () => {
    43→      setIsScrolled(window.scrollY > 50);
    44→    };
    45→    
    46→    window.addEventListener('scroll', handleScroll, { passive: true });
    47→    return () => window.removeEventListener('scroll', handleScroll);
    48→  }, []);
    49→
    50→  const navItems = [
    51→    { label: 'О нас', href: '#about' },
    52→    { label: 'Меню', href: '#menu' },
    53→    { label: 'Услуги', href: '#services' },
    54→    { label: 'Галерея', href: '#gallery' },
    55→    { label: 'Отзывы', href: '#testimonials' },
    56→    { label: 'Контакты', href: '#contact' },
    57→  ];
    58→
    59→  return (
    60→    <>
    61→      <motion.header
    62→        style={{ backgroundColor: headerBg, backdropFilter: headerBlur }}
    63→        className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
    64→          isScrolled ? 'py-3 shadow-lg' : 'py-6'
    65→        }`}
    66→      >
    67→        <nav className='container mx-auto px-4 md:px-6 flex items-center justify-between'>
    68→          {/* Logo */}
    69→          <Link href='/' className='relative z-10'>
    70→            <span className='font-display text-2xl font-bold'>Название</span>
    71→          </Link>
    72→
    73→          {/* Desktop Nav */}
    74→          <div className='hidden lg:flex items-center gap-8'>
    75→            {navItems.map((item) => (
    76→              <Link
    77→                key={item.href}
    78→                href={item.href}
    79→                className='text-sm uppercase tracking-wider text-cream/80 hover:text-bordeaux transition-colors relative group'
    80→              >
    81→                {item.label}
    82→                <span className='absolute -bottom-1 left-0 w-0 h-0.5 bg-bordeaux transition-all duration-300 group-hover:w-full' />
    83→              </Link>
    84→            ))}
    85→            <MagneticButton className='bg-bordeaux text-white px-6 py-3 rounded-full text-sm font-semibold'>
    86→              Рассчитать
    87→            </MagneticButton>
    88→          </div>
    89→
    90→          {/* Mobile Menu Toggle */}
    91→          <button
    92→            onClick={() => setIsMobileMenuOpen(!isMobileMenuOpen)}
    93→            className='lg:hidden relative z-10 w-10 h-10 flex items-center justify-center'
    94→            aria-label={isMobileMenuOpen ? 'Закрыть меню' : 'Открыть меню'}
    95→          >
    96→            <div className='flex flex-col gap-1.5'>
    97→              <motion.span
    98→                animate={isMobileMenuOpen ? { rotate: 45, y: 6 } : { rotate: 0, y: 0 }}
    99→                className='w-6 h-0.5 bg-current block'
   100→              />
   101→              <motion.span
   102→                animate={isMobileMenuOpen ? { opacity: 0 } : { opacity: 1 }}
   103→                className='w-6 h-0.5 bg-current block'
   104→              />
   105→              <motion.span
   106→                animate={isMobileMenuOpen ? { rotate: -45, y: -6 } : { rotate: 0, y: 0 }}
   107→                className='w-6 h-0.5 bg-current block'
   108→              />
   109→            </div>
   110→          </button>
   111→        </nav>
   112→      </motion.header>
   113→
   114→      {/* Full-screen Mobile Menu */}
   115→      <AnimatePresence>
   116→        {isMobileMenuOpen && (
   117→          <motion.div
   118→            initial={{ opacity: 0 }}
   119→            animate={{ opacity: 1 }}
   120→            exit={{ opacity: 0 }}
   121→            className='fixed inset-0 z-40 bg-night flex flex-col items-center justify-center gap-8'
   122→          >
   123→            {navItems.map((item, index) => (
   124→              <motion.div
   125→                key={item.href}
   126→                initial={{ opacity: 0, y: 20 }}
   127→                animate={{ opacity: 1, y: 0 }}
   128→                exit={{ opacity: 0, y: 20 }}
   129→                transition={{ delay: index * 0.1 }}
   130→              >
   131→                <Link
   132→                  href={item.href}
   133→                  onClick={() => setIsMobileMenuOpen(false)}
   134→                  className='text-4xl font-display text-cream hover:text-bordeaux transition-colors'
   135→                >
   136→                  {item.label}
   137→                </Link>
   138→              </motion.div>
   139→            ))}
   140→          </motion.div>
   141→        )}
   142→      </AnimatePresence>
   143→    </>
   144→  );
   145→}
   146→```

   148→### 2. Chapter Navigation (Progress Indicator)
   149→
   150→```tsx
   151→// components/catering/chapter-nav.tsx (enhanced)
   152→'use client';
   153→
   154→import { useEffect, useRef, useState } from 'react';
   155→import { motion, useScroll, useSpring } from 'framer-motion';
   156→
   157→interface Chapter {
   158→  id: string;
   159→  label: string;
   160→}
   161→
   162→const chapters: Chapter[] = [
   163→  { id: 'hero', label: 'Главная' },
   164→  { id: 'about', label: 'О нас' },
   165→  { id: 'menu', label: 'Меню' },
   166→  { id: 'services', label: 'Услуги' },
   167→  { id: 'gallery', label: 'Галерея' },
   168→  { id: 'testimonials', label: 'Отзывы' },
   169→  { id: 'calculator', label: 'Калькулятор' },
   170→  { id: 'contact', label: 'Контакты' },
   171→];
   172→
   173→export function ChapterNav() {
   174→  const [activeChapter, setActiveChapter] = useState(0);
   175→  const navRef = useRef<HTMLElement>(null);
   176→
   177→  const { scrollYProgress } = useScroll({
   178→    offset: ['start start', 'end end'],
   179→  });
   180→
   181→  const smoothProgress = useSpring(scrollYProgress, {
   182→    stiffness: 100,
   183→    damping: 30,
   184→    restDelta: 0.001,
   185→  });
   186→
   187→  useEffect(() => {
   188→    const observers: IntersectionObserver[] = [];
   189→
   190→    chapters.forEach((chapter) => {
   191→      const element = document.getElementById(chapter.id);
   192→      if (!element) return;
   193→
   194→      const observer = new IntersectionObserver(
   195→        ([entry]) => {
   196→          if (entry.isIntersecting) {
   197→            setActiveChapter(chapters.indexOf(chapter));
   198→          }
   199→        },
   200→        { threshold: 0.5, rootMargin: '-20% 0px -20% 0px' }
   201→      );
   202→
   203→      observer.observe(element);
   204→      observers.push(observer);
   205→    });
   206→
   207→    return () => observers.forEach((obs) => obs.disconnect());
   208→  }, []);
   209→
   210→  const scrollToChapter = (index: number) => {
   211→    const element = document.getElementById(chapters[index].id);
   212→    element?.scrollIntoView({ behavior: 'smooth' });
   213→  };
   214→
   215→  return (
   216→    <motion.nav
   217→      ref={navRef}
   218→      initial={{ x: 50, opacity: 0 }}
   219→      animate={{ x: 0, opacity: 1 }}
   220→      transition={{ delay: 1, duration: 0.5 }}
   221→      className='fixed right-6 top-1/2 -translate-y-1/2 z-40 hidden xl:flex flex-col items-center gap-4'
   222→    >
   223→      {/* Progress line */}
   224→      <div className='absolute top-0 bottom-0 w-0.5 bg-sand/30 -z-10'>
   225→        <motion.div
   226→          className='w-full bg-bordeaux'
   227→          style={{ height: smoothProgress }}
   228→        />
   229→      </div>
   230→
   231→      {chapters.map((chapter, index) => (
   232→        <button
   233→          key={chapter.id}
   234→          onClick={() => scrollToChapter(index)}
   235→          className={`group relative flex items-center justify-center w-8 h-8 rounded-full transition-all duration-300 ${
   236→            activeChapter === index
   237→              ? 'bg-bordeaux text-white'
   238→              : 'bg-transparent text-muted hover:text-foreground'
   239→          }`}
   240→          aria-label={`Перейти к разделу ${chapter.label}`}
   241→        >
   242→          <span className='text-xs font-mono'>{String(index + 1).padStart(2, '0')}</span>
   243→          
   244→          {/* Tooltip on hover */}
   245→          <span className='absolute right-full mr-4 px-3 py-1.5 bg-charcoal text-cream text-xs rounded opacity-0 group-hover:opacity-100 transition-opacity whitespace-nowrap pointer-events-none'>
   246→            {chapter.label}
   247→          </span>
   248→        </button>
   249→      ))}
   250→    </motion.nav>
   251→  );
   252→}
   253→```

   257→### 3. Interactive Gallery with Lightbox
   258→
   259→```tsx
   260→// components/catering/events-gallery.tsx (enhanced)
   261→'use client';
   262→
   263→import { useState, useCallback } from 'react';
   264→import { motion, AnimatePresence } from 'framer-motion';
   265→import Image from 'next/image';
   266→
   267→interface GalleryItem {
   268→  src: string;
   269→  alt: string;
   270→  category: string;
   271→  title?: string;
   272→}
   273→
   274→export function EventsGallery({ items }: { items: GalleryItem[] }) {
   275→  const [activeFilter, setActiveFilter] = useState<string>('all');
   276→  const [lightboxIndex, setLightboxIndex] = useState<number | null>(null);
   277→  const [hoveredIndex, setHoveredIndex] = useState<number | null>(null);
   278→
   279→  const categories = ['all', ...new Set(items.map((i) => i.category))];
   280→
   281→  const filteredItems =
   282→    activeFilter === 'all'
   283→      ? items
   284→      : items.filter((i) => i.category === activeFilter);
   285→
   286→  const openLightbox = useCallback((index: number) => {
   287→    setLightboxIndex(index);
   288→    document.body.style.overflow = 'hidden';
   289→  }, []);
   290→
   291→  const closeLightbox = useCallback(() => {
   292→    setLightboxIndex(null);
   293→    document.body.style.overflow = '';
   294→  }, []);
   295→
   296→  const navigateLightbox = (direction: 'prev' | 'next') => {
   297→    if (lightboxIndex === null) return;
   298→    const newIndex =
   299→      direction === 'prev'
   300→        ? (lightboxIndex - 1 + filteredItems.length) % filteredItems.length
   301→        : (lightboxIndex + 1) % filteredItems.length;
   302→    setLightboxIndex(newIndex);
   303→  };
   304→
   305→  return (
   306→    <>
   307→      {/* Filter tabs */}
   308→      <div className='flex flex-wrap justify-center gap-3 mb-12'>
   309→        {categories.map((category) => (
   310→          <button
   311→            key={category}
   312→            onClick={() => setActiveFilter(category)}
   313→            className={`px-5 py-2 rounded-full text-sm capitalize transition-all duration-300 ${
   314→              activeFilter === category
   315→                ? 'bg-bordeaux text-white'
   316→                : 'bg-parchment text-espresso hover:bg-sand'
   317→            }`}
   318→          >
   319→            {category === 'all' ? 'Все' : category}
   320→          </button>
   321→        ))}
   322→      </div>
   323→
   324→      {/* Grid */}
   325→      <motion.div
   326→        layout
   327→        className='grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4'
   328→      >
   329→        <AnimatePresence mode='popLayout'>
   330→          {filteredItems.map((item, index) => (
   331→            <motion.div
   332→              key={item.src}
   333→              layout
   334→              initial={{ opacity: 0, scale: 0.9 }}
   335→              animate={{ opacity: 1, scale: 1 }}
   336→              exit={{ opacity: 0, scale: 0.9 }}
   337→              transition={{ duration: 0.3 }}
   338→              className='group relative aspect-[4/3] overflow-hidden rounded-2xl cursor-pointer'
   339→              onMouseEnter={() => setHoveredIndex(index)}
   340→              onMouseLeave={() => setHoveredIndex(null)}
   341→              onClick={() => openLightbox(index)}
   342→            >
   343→              <Image
   344→                src={item.src}
   345→                alt={item.alt}
   346→                fill
   347→                className='object-cover transition-transform duration-700 group-hover:scale-110'
   348→              />
   349→
   350→              {/* Overlay */}
   351→              <div
   352→                className={`absolute inset-0 bg-gradient-to-t from-night via-night/40 to-transparent 
   353→                  flex items-end p-6 transition-opacity duration-300 ${
   354→                    hoveredIndex === index ? 'opacity-100' : 'opacity-0'
   355→                  }`}
   356→              >
   357→                <div>
   358→                  {item.title && (
   359→                    <h3 className='text-white font-display text-xl'>{item.title}</h3>
   360→                  )}
   361→                  <p className='text-cream/70 text-sm capitalize'>{item.category}</p>
   362→                </div>
   363→              </div>
   364→
   365→              {/* Zoom icon */}
   366→              <div
   367→                className={`absolute top-4 right-4 w-10 h-10 rounded-full bg-white/20 backdrop-blur-sm
   368→                  flex items-center justify-center transition-all duration-300 ${
   369→                    hoveredIndex === index
   370→                      ? 'opacity-100 translate-y-0'
   371→                      : 'opacity-0 -translate-y-2'
   372→                  }`}
   373→              >
   374→                <svg className='w-5 h-5 text-white' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   375→                  <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0zM10 7v6m3-3H7' />
   376→                </svg>
   377→              </div>
   378→            </motion.div>
   379→          ))}
   380→        </AnimatePresence>
   381→      </motion.div>
   382→
   383→      {/* Lightbox */}
   384→      <AnimatePresence>
   385→        {lightboxIndex !== null && (
   386→          <motion.div
   387→            initial={{ opacity: 0 }}
   388→            animate={{ opacity: 1 }}
   389→            exit={{ opacity: 0 }}
   390→            className='fixed inset-0 z-[100] bg-night/95 flex items-center justify-center p-4'
   391→            onClick={closeLightbox}
   392→          >
   393→            {/* Close button */}
   394→            <button
   395→              onClick={closeLightbox}
   396→              className='absolute top-6 right-6 w-12 h-12 rounded-full bg-white/10 flex items-center justify-center text-white hover:bg-white/20 transition-colors z-10'
   397→              aria-label='Закрыть'
   398→            >
   399→              <svg className='w-6 h-6' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   400→                <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M6 18L18 6M6 6l12 12' />
   401→              </svg>
   402→            </button>
   403→
   404→            {/* Image */}
   405→            <motion.div
   406→              key={lightboxIndex}
   407→              initial={{ opacity: 0, scale: 0.9 }}
   408→              animate={{ opacity: 1, scale: 1 }}
   409→              exit={{ opacity: 0, scale: 0.9 }}
   410→              className='relative max-w-5xl max-h-[80vh] aspect-video'
   411→              onClick={(e) => e.stopPropagation()}
   412→            >
   413→              <Image
   414→                src={filteredItems[lightboxIndex].src}
   415→                alt={filteredItems[lightboxIndex].alt}
   416→                fill
   417→                className='object-contain'
   418→              />
   419→
   420→              {/* Caption */}
   421→              <div className='absolute bottom-0 left-0 right-0 p-6 bg-gradient-to-t from-night to-transparent'>
   422→                <h3 className='text-white font-display text-xl'>
   423→                  {filteredItems[lightboxIndex].title || filteredItems[lightboxIndex].alt}
   424→                </h3>
   425→              </div>
   426→            </motion.div>
   427→
   428→            {/* Navigation arrows */}
   429→            <button
   430→              onClick={(e) => {
   431→                e.stopPropagation();
   432→                navigateLightbox('prev');
   433→              }}
   434→              className='absolute left-4 top-1/2 -translate-y-1/2 w-12 h-12 rounded-full bg-white/10 flex items-center justify-center text-white hover:bg-white/20 transition-colors'
   435→              aria-label='Предыдущее изображение'
   436→            >
   437→              <svg className='w-6 h-6' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   438→                <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M15 19l-7-7 7-7' />
   439→              </svg>
   440→            </button>
   441→            <button
   442→              onClick={(e) => {
   443→                e.stopPropagation();
   444→                navigateLightbox('next');
   445→              }}
   446→              className='absolute right-4 top-1/2 -translate-y-1/2 w-12 h-12 rounded-full bg-white/10 flex items-center justify-center text-white hover:bg-white/20 transition-colors'
   447→              aria-label='Следующее изображение'
   448→            >
   449→              <svg className='w-6 h-6' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   450→                <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M9 5l7 7-7 7' />
   451→              </svg>
   452→            </button>
   453→
   454→            {/* Counter */}
   455→            <div className='absolute bottom-6 left-1/2 -translate-x-1/2 text-white/60 text-sm'>
   456→              {lightboxIndex + 1} / {filteredItems.length}
   457→            </div>
   458→          </motion.div>
   459→        )}
   460→      </AnimatePresence>
   461→    </>
   462→  );
   463→}
   464→```

   466→### 4. Testimonial Carousel (Ridgewells style)
   467→
   468→```tsx
   469→// components/catering/testimonials.tsx (enhanced)
   470→'use client';
   471→
   472→import { useState, useEffect, useCallback } from 'react';
   473→import { motion, AnimatePresence } from 'framer-motion';
   474→import { Star, ChevronLeft, ChevronRight, Quote } from 'lucide-react';
   475→
   476→interface Testimonial {
   477→  company: string;
   478→  text: string;
   479→  author: string;
   480→  role: string;
   481→  rating: number;
   482→  image?: string;
   483→}
   484→
   485→export function TestimonialCarousel({ testimonials }: { testimonials: Testimonial[] }) {
   486→  const [current, setCurrent] = useState(0);
   487→  const [direction, setDirection] = useState(0);
   488→
   489→  const next = useCallback(() => {
   490→    setDirection(1);
   491→    setCurrent((prev) => (prev + 1) % testimonials.length);
   492→  }, [testimonials.length]);
   493→
   494→  const prev = useCallback(() => {
   495→    setDirection(-1);
   496→    setCurrent((prev) => (prev - 1 + testimonials.length) % testimonials.length);
   497→  }, [testimonials.length]);
   498→
   499→  // Auto-advance
   500→  useEffect(() => {
   501→    const timer = setInterval(next, 8000);
   502→    return () => clearInterval(timer);
   503→  }, [next]);
   504→
   505→  const variants = {
   506→    enter: (direction: number) => ({
   507→      x: direction > 0 ? 300 : -300,
   508→      opacity: 0,
   509→    }),
   510→    center: {
   511→      x: 0,
   512→      opacity: 1,
   513→    }),
   514→    exit: (direction: number) => ({
   515→      x: direction < 0 ? 300 : -300,
   516→      opacity: 0,
   517→    }),
   518→  };
   519→
   520→  return (
   521→    <div className='relative max-w-4xl mx-auto'>
   522→      {/* Quote icon */}
   523→      <div className='absolute -top-8 left-1/2 -translate-x-1/2 text-bordeaux/20'>
   524→        <Quote className='w-16 h-16' />
   525→      </div>
   526→
   527→      {/* Carousel */}
   528→      <div className='overflow-hidden min-h-[300px] flex items-center'>
   529→        <AnimatePresence mode='wait' custom={direction}>
   530→          <motion.div
   531→            key={current}
   532→            custom={direction}
   533→            variants={variants}
   534→            initial='enter'
   535→            animate='center'
   536→            exit='exit'
   537→            transition={{
   538→              x: { type: 'spring', stiffness: 200, damping: 25 },
   539→              opacity: { duration: 0.3 },
   540→            }}
   541→            className='w-full text-center px-8'
   542→          >
   543→            {/* Stars */}
   544→            <div className='flex justify-center gap-1 mb-6'>
   545→              {Array.from({ length: 5 }).map((_, i) => (
   546→                <Star
   547→                  key={i}
   548→                  className={`w-5 h-5 ${
   549→                    i < testimonials[current].rating
   550→                      ? 'fill-gold text-gold'
   551→                      : 'text-sand'
   552→                  }`}
   553→                />
   554→              ))}
   555→            </div>
   556→
   557→            {/* Text */}
   558→            <blockquote className='text-xl md:text-2xl font-light leading-relaxed text-cream/90 mb-8 italic'>
   559→              &ldquo;{testimonials[current].text}&rdquo;
   560→            </blockquote>
   561→
   562→            {/* Author */}
   563→            <div className='flex items-center justify-center gap-4'>
   564→              {testimonials[current].image && (
   565→                <div className='w-14 h-14 rounded-full overflow-hidden'>
   566→                  <Image
   567→                    src={testimonials[current].image!}
   568→                    alt={testimonials[current].author}
   569→                    width={56}
   570→                    height={56}
   571→                    className='object-cover'
   572→                  />
   573→                </div>
   574→              )}
   575→              <div className='text-left'>
   576→                <cite className='not-italic font-semibold text-white'>
   577→                  {testimonials[current].author}
   578→                </cite>
   579→                <p className='text-sm text-cream/60'>
   580→                  {testimonials[current].role}, {testimonials[current].company}
   581→                </p>
   582→              </div>
   583→            </div>
   584→          </motion.div>
   585→        </AnimatePresence>
   586→      </div>
   587→
   588→      {/* Navigation */}
   589→      <div className='flex items-center justify-center gap-6 mt-8'>
   590→        <button
   591→          onClick={prev}
   592→          className='w-12 h-12 rounded-full border border-sand/30 flex items-center justify-center text-cream/60 hover:text-bordeau hover:border-bordeaux transition-colors'
   593→          aria-label='Предыдущий отзыв'
   594→        >
   595→          <ChevronLeft className='w-5 h-5' />
   596→        </button>
   597→
   598→        {/* Dots */}
   599→        <div className='flex gap-2'>
   600→          {testimonials.map((_, i) => (
   601→            <button
   602→              key={i}
   603→              onClick={() => {
   604→                setDirection(i > current ? 1 : -1);
   605→                setCurrent(i);
   606→              }}
   607→              className={`w-2 h-2 rounded-full transition-all duration-300 ${
   608→                i === current ? 'w-8 bg-bordeaux' : 'bg-sand/30 hover:bg-sand/60'

   615→        <button
   616→          onClick={next}
   617→          className='w-12 h-12 rounded-full border border-sand/30 flex items-center justify-center text-cream/60 hover:text-bordeau hover:border-bordeaux transition-colors'
   618→          aria-label='Следующий отзыв'
   619→        >
   620→          <ChevronRight className='w-5 h-5' />

   632→```tsx
   633→// components/catering/contact-form.tsx (enhanced)
   634→'use client';
   635→
   636→import { useState } from 'react';
   637→import { motion, AnimatePresence } from 'framer-motion';
   638→import { Send, CheckCircle, Loader2 } from 'lucide-react';
   639→
   640→interface FormData {
   641→  name: string;
   642→  phone: string;
   643→  email: string;
   644→  eventType: string;
   645→  guests: string;
   646→  date: string;
   647→  message: string;
   648→}
   649→
   650→type FormStatus = 'idle' | 'submitting' | 'success' | 'error';
   651→
   652→export function ContactForm() {
   653→  const [formData, setFormData] = useState<FormData>({
   654→    name: '',
   655→    phone: '',
   656→    email: '',
   657→    eventType: '',
   658→    guests: '',
   659→    date: '',
   660→    message: '',
   661→  });
   662→  const [status, setStatus] = useState<FormStatus>('idle');
   663→  const [focusedField, setFocusedField] = useState<string | null>(null);
   664→
   665→  const handleSubmit = async (e: React.FormEvent) => {
   666→    e.preventDefault();
   667→    setStatus('submitting');
   668→
   669→    try {
   670→      const response = await fetch('/api/lead', {
   671→        method: 'POST',
   672→        headers: { 'Content-Type': 'application/json' },
   673→        body: JSON.stringify(formData),
   674→      });
   675→
   676→      if (response.ok) {
   677→        setStatus('success');
   678→      } else {
   679→        setStatus('error');
   680→      }
   681→    } catch {
   682→      setStatus('error');
   683→    }
   684→  };
   685→
   686→  const inputClasses = (field: string) =>
   687→    `w-full pt-6 pb-2 bg-transparent border-b-2 transition-colors duration-300 outline-none
   688→     ${focusedField === field ? 'border-bordeaux' : 'border-sand'}
   689→     ${status === 'success' ? 'border-success' : ''}`;
   690→
   691→  return (
   692→    <form onSubmit={handleSubmit} className='max-w-lg mx-auto space-y-8'>
   693→      <AnimatePresence mode='wait'>
   694→        {status === 'success' ? (
   695→          <motion.div
   696→            key='success'
   697→            initial={{ opacity: 0, y: 20 }}
   698→            animate={{ opacity: 1, y: 0 }}
   699→            exit={{ opacity: 0, y: -20 }}
   700→            className='text-center py-16'
   701→          >
   702→            <CheckCircle className='w-16 h-16 text-success mx-auto mb-4' />
   703→            <h3 className='text-2xl font-display mb-2'>Заявка отправлена!</h3>
   704→            <p className='text-muted'>Мы свяжемся с вами в ближайшее время</p>
   705→          </motion.div>
   706→        ) : (
   707→          <motion.div key='form'>
   708→            {/* Name */}
   709→            <div className='relative'>
   710→              <input
   711→                type='text'
   712→                value={formData.name}
   713→                onChange={(e) => setFormData({ ...formData, name: e.target.value })}
   714→                onFocus={() => setFocusedField('name')}
   715→                onBlur={() => setFocusedField(null)}
   716→                placeholder=' '
   717→                className={inputClasses('name')}
   718→                required
   719→              />
   720→              <label
   721→                className={`absolute left-0 transition-all duration-300 pointer-events-none ${
   722→                  focusedField === 'name || formData.name
   723→                    ? 'top-1 text-xs text-bordeaux'
   724→                    : 'top-4 text-muted'
   725→                }`}
   726→              >
   727→                Ваше имя *
   728→              </label>
   729→            </div>
   730→
   731→            {/* Phone */}
   732→            <div className='relative'>
   733→              <input
   734→                type='tel'
   735→                value={formData.phone}
   736→                onChange={(e) => setFormData({ ...formData, phone: e.target.value })}
   737→                onFocus={() => setFocusedField('phone')}
   738→                onBlur={() => setFocusedField(null)}
   739→                placeholder=' '
   740→                className={inputClasses('phone')}
   741→                required
   742→              />
   743→              <label
   744→                className={`absolute left-0 transition-all duration-300 pointer-events-none ${
   745→                  focusedField === 'phone || formData.phone
   746→                    ? 'top-1 text-xs text-bordeaux'
   747→                    : 'top-4 text-muted'
   748→                }`}
   749→              >
   750→                Телефон *
   751→              </label>
   752→            </div>
   753→
   754→            {/* Event Type */}
   755→            <div className='relative'>
   756→              <select
   757→                value={formData.eventType}
   758→                onChange={(e) => setFormData({ ...formData, eventType: e.target.value })}
   759→                onFocus={() => setFocusedField('eventType')}
   760→                onBlur={() => setFocusedField(null)}
   761→                className={`${inputClasses('eventType')} appearance-none cursor-pointer`}
   762→                required
   763→              >
   764→                <option value='' disabled>Тип мероприятия</option>
   765→                <option value='banquet'>Банкет</option>
   766→                <option value='furshet'>Фуршет</option>
   767→                <option value='wedding'>Свадьба</option>
   768→                <option value='corporate'>Корпоратив</option>
   769→                <option value='other'>Другое</option>
   770→              </select>
   771→              <label
   772→                className={`absolute left-0 transition-all duration-300 pointer-events-none ${
   773→                  focusedField === 'eventType || formData.eventType
   774→                    ? 'top-1 text-xs text-bordeaux'
   775→                    : 'top-4 text-muted'
   776→                }`}
   777→              >
   778→              Тип мероприятия *
   779→              </label>
   780→            </div>
   781→
   782→            {/* Submit Button */}
   783→            <motion.button
   784→              type='submit'
   785→              disabled={status === 'submitting'}
   786→              className='w-full py-4 bg-bordeaux text-white rounded-full font-semibold
   787→                flex items-center justify-center gap-2
   788→                hover:bg-bordeaux-dark transition-colors disabled:opacity-70'
   789→              whileHover={{ scale: 1.02 }}
   790→              whileTap={{ scale: 0.98 }}

   792→              {status === 'submitting' ? (
   793→                <>
   794→                  <Loader2 className='w-5 h-5 animate-spin' />
   795→                  Отправка...
   796→                </>
   797→              ) : (
   798→                <>
   799→                  <Send className='w-5 h-5' />
   800→                  Отправить заявку
   801→                </>
   802→              )}
   803→            </motion.button>
   804→          </motion.div>
   805→        )}
   806→      </AnimatePresence>
   807→    </form>
   808→  );
   809→}

   812→### 6. Accordion / FAQ Component
   813→
   814→```tsx
   815→// components/ui/accordion.tsx (enhanced)
   816→'use client';
   817→
   818→import { useState } from 'react';
   819→import { motion, AnimatePresence } from 'framer-motion';
   820→import { ChevronDown } from 'lucide-react';
   821→
   822→interface AccordionItem {
   823→  question: string;
   824→  answer: string;
   825→}
   826→
   827→export function Accordion({ items }: { items: AccordionItem[] }) {
   828→  const [openIndex, setOpenIndex] = useState<number | null>(null);
   829→
   830→  const toggle = (index: number) => {
   831→    setOpenIndex(openIndex === index ? null : index);
   832→  };
   833→
   834→  return (
   835→    <div className='space-y-3'>
   836→      {items.map((item, index) => (
   837→        <div
   838→          key={index}
   839→          className='rounded-2xl overflow-hidden bg-card border border-border'
   840→        >
   841→          <button
   842→            onClick={() => toggle(index)}
   843→            className='w-full flex items-center justify-between p-6 text-left'
   844→            aria-expanded={openIndex === index}
   845→          >
   846→            <span className='font-medium pr-4'>{item.question}</span>
   847→            <motion.span
   848→              animate={{ rotate: openIndex === index ? 180 : 0 }}
   849→              transition={{ duration: 0.3 }}
   850→              className='flex-shrink-0'
   851→            >
   852→              <ChevronDown className='w-5 h-5 text-muted' />
   853→            </motion.span>
   854→          </button>
   855→
   856→          <AnimatePresence>
   857→            {openIndex === index && (
   858→              <motion.div
   859→                initial={{ height: 0, opacity: 0 }}
   860→                animate={{ height: 'auto', opacity: 1 }}
   861→                exit={{ height: 0, opacity: 0 }}
   862→                transition={{ duration: 0.3, ease: 'easeInOut' }}
   863→              >
   864→                <div className='px-6 pb-6 text-muted'>
   865→                  {item.answer}
   866→                </div>
   867→              </motion.div>
   868→            )}
   869→          </AnimatePresence>
   870→        </div>
   871→      ))}
   872→    </div>
   873→  );
   874→}

   904→### 8. Back to Top Button
   905→
   906→```tsx
   907→// components/motion/back-to-top.tsx
   908→'use client';
   909→
   910→import { useState, useEffect } from 'react';
   911→import { motion, AnimatePresence } from 'framer-motion';
   912→import { ArrowUp } from 'lucide-react';
   913→
   914→export function BackToTop() {
   915→  const [isVisible, setIsVisible] = useState(false);
   916→
   917→  useEffect(() => {
   918→    const toggleVisibility = () => {
   919→      setIsVisible(window.scrollY > 500);
   920→    };
   921→
   922→    window.addEventListener('scroll', toggleVisibility, { passive: true });
   923→    return () => window.removeEventListener('scroll', toggleVisibility);
   924→  }, []);
   925→
   926→  const scrollToTop = () => {
   927→    window.scrollTo({ top: 0, behavior: 'smooth' });
   928→  };
   929→
   930→  return (
   931→    <AnimatePresence>
   932→      {isVisible && (
   933→        <motion.button
   934→          initial={{ opacity: 0, y: 20 }}
   935→          animate={{ opacity: 1, y: 0 }}
   936→          exit={{ opacity: 0, y: 20 }}
   937→          onClick={scrollToTop}
   938→          className='fixed bottom-8 right-8 z-50 w-12 h-12 rounded-full bg-bordeaux text-white
   939→            flex items-center justify-center shadow-lg hover:bg-bordeaux-dark transition-colors'
   940→          aria-label='Наверх'
   941→        >
   942→          <ArrowUp className='w-5 h-5' />
   943→        </motion.button>
   944→      )}
   945→    </AnimatePresence>
   946→  );
   947→}
   948→```
