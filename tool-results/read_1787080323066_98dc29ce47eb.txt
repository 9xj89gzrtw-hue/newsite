    15→    15→### 1. Sticky Header with Scroll Effect (M Culinary style)
    16→    16→
    17→    17→```tsx
    18→    18→// components/catering/site-header.tsx (enhanced)
    19→    19→'use client';
    20→    20→
    21→    21→import { useState, useEffect } from 'react';
    22→    22→import { motion, useScroll, useTransform } from 'framer-motion';
    23→    23→import Link from 'next/link';
    24→    24→
    25→    25→export function SiteHeader() {
    26→    26→  const [isScrolled, setIsScrolled] = useState(false);
    27→    27→  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);
    28→    28→  
    29→    29→  const { scrollY } = useScroll();
    30→    30→  const headerBg = useTransform(
    31→    31→    scrollY,
    32→    32→    [0, 100],
    33→    33→    ['rgba(26, 22, 20, 0)', 'rgba(26, 22, 20, 0.95)']
    34→    34→  );
    35→    35→  const headerBlur = useTransform(
    36→    36→    scrollY,
    37→    37→    [0, 100],
    38→    38→    ['blur(0px)', 'blur(20px)']
    39→    39→  );
    40→    40→
    41→    41→  useEffect(() => {
    42→    42→    const handleScroll = () => {
    43→    43→      setIsScrolled(window.scrollY > 50);
    44→    44→    };
    45→    45→    
    46→    46→    window.addEventListener('scroll', handleScroll, { passive: true });
    47→    47→    return () => window.removeEventListener('scroll', handleScroll);
    48→    48→  }, []);
    49→    49→
    50→    50→  const navItems = [
    51→    51→    { label: 'О нас', href: '#about' },
    52→    52→    { label: 'Меню', href: '#menu' },
    53→    53→    { label: 'Услуги', href: '#services' },
    54→    54→    { label: 'Галерея', href: '#gallery' },
    55→    55→    { label: 'Отзывы', href: '#testimonials' },
    56→    56→    { label: 'Контакты', href: '#contact' },
    57→    57→  ];
    58→    58→
    59→    59→  return (
    60→    60→    <>
    61→    61→      <motion.header
    62→    62→        style={{ backgroundColor: headerBg, backdropFilter: headerBlur }}
    63→    63→        className={`fixed top-0 left-0 right-0 z-50 transition-all duration-300 ${
    64→    64→          isScrolled ? 'py-3 shadow-lg' : 'py-6'
    65→    65→        }`}
    66→    66→      >
    67→    67→        <nav className='container mx-auto px-4 md:px-6 flex items-center justify-between'>
    68→    68→          {/* Logo */}
    69→    69→          <Link href='/' className='relative z-10'>
    70→    70→            <span className='font-display text-2xl font-bold'>Название</span>
    71→    71→          </Link>
    72→    72→
    73→    73→          {/* Desktop Nav */}
    74→    74→          <div className='hidden lg:flex items-center gap-8'>
    75→    75→            {navItems.map((item) => (
    76→    76→              <Link
    77→    77→                key={item.href}
    78→    78→                href={item.href}
    79→    79→                className='text-sm uppercase tracking-wider text-cream/80 hover:text-bordeaux transition-colors relative group'
    80→    80→              >
    81→    81→                {item.label}
    82→    82→                <span className='absolute -bottom-1 left-0 w-0 h-0.5 bg-bordeaux transition-all duration-300 group-hover:w-full' />
    83→    83→              </Link>
    84→    84→            ))}
    85→    85→            <MagneticButton className='bg-bordeaux text-white px-6 py-3 rounded-full text-sm font-semibold'>
    86→    86→              Рассчитать
    87→    87→            </MagneticButton>
    88→    88→          </div>
    89→    89→
    90→    90→          {/* Mobile Menu Toggle */}
    91→    91→          <button
    92→    92→            onClick={() => setIsMobileMenuOpen(!isMobileMenuOpen)}
    93→    93→            className='lg:hidden relative z-10 w-10 h-10 flex items-center justify-center'
    94→    94→            aria-label={isMobileMenuOpen ? 'Закрыть меню' : 'Открыть меню'}
    95→    95→          >
    96→    96→            <div className='flex flex-col gap-1.5'>
    97→    97→              <motion.span
    98→    98→                animate={isMobileMenuOpen ? { rotate: 45, y: 6 } : { rotate: 0, y: 0 }}
    99→    99→                className='w-6 h-0.5 bg-current block'
   100→   100→              />
   101→   101→              <motion.span
   102→   102→                animate={isMobileMenuOpen ? { opacity: 0 } : { opacity: 1 }}
   103→   103→                className='w-6 h-0.5 bg-current block'
   104→   104→              />
   105→   105→              <motion.span
   106→   106→                animate={isMobileMenuOpen ? { rotate: -45, y: -6 } : { rotate: 0, y: 0 }}
   107→   107→                className='w-6 h-0.5 bg-current block'
   108→   108→              />
   109→   109→            </div>
   110→   110→          </button>
   111→   111→        </nav>
   112→   112→      </motion.header>
   113→   113→
   114→   114→      {/* Full-screen Mobile Menu */}
   115→   115→      <AnimatePresence>
   116→   116→        {isMobileMenuOpen && (
   117→   117→          <motion.div
   118→   118→            initial={{ opacity: 0 }}
   119→   119→            animate={{ opacity: 1 }}
   120→   120→            exit={{ opacity: 0 }}
   121→   121→            className='fixed inset-0 z-40 bg-night flex flex-col items-center justify-center gap-8'
   122→   122→          >
   123→   123→            {navItems.map((item, index) => (
   124→   124→              <motion.div
   125→   125→                key={item.href}
   126→   126→                initial={{ opacity: 0, y: 20 }}
   127→   127→                animate={{ opacity: 1, y: 0 }}
   128→   128→                exit={{ opacity: 0, y: 20 }}
   129→   129→                transition={{ delay: index * 0.1 }}
   130→   130→              >
   131→   131→                <Link
   132→   132→                  href={item.href}
   133→   133→                  onClick={() => setIsMobileMenuOpen(false)}
   134→   134→                  className='text-4xl font-display text-cream hover:text-bordeaux transition-colors'
   135→   135→                >
   136→   136→                  {item.label}
   137→   137→                </Link>
   138→   138→              </motion.div>
   139→   139→            ))}
   140→   140→          </motion.div>
   141→   141→        )}
   142→   142→      </AnimatePresence>
   143→   143→    </>
   144→   144→  );
   145→   145→}
   146→   146→```

   148→   148→### 2. Chapter Navigation (Progress Indicator)
   149→   149→
   150→   150→```tsx
   151→   151→// components/catering/chapter-nav.tsx (enhanced)
   152→   152→'use client';
   153→   153→
   154→   154→import { useEffect, useRef, useState } from 'react';
   155→   155→import { motion, useScroll, useSpring } from 'framer-motion';
   156→   156→
   157→   157→interface Chapter {
   158→   158→  id: string;
   159→   159→  label: string;
   160→   160→}
   161→   161→
   162→   162→const chapters: Chapter[] = [
   163→   163→  { id: 'hero', label: 'Главная' },
   164→   164→  { id: 'about', label: 'О нас' },
   165→   165→  { id: 'menu', label: 'Меню' },
   166→   166→  { id: 'services', label: 'Услуги' },
   167→   167→  { id: 'gallery', label: 'Галерея' },
   168→   168→  { id: 'testimonials', label: 'Отзывы' },
   169→   169→  { id: 'calculator', label: 'Калькулятор' },
   170→   170→  { id: 'contact', label: 'Контакты' },
   171→   171→];
   172→   172→
   173→   173→export function ChapterNav() {
   174→   174→  const [activeChapter, setActiveChapter] = useState(0);
   175→   175→  const navRef = useRef<HTMLElement>(null);
   176→   176→
   177→   177→  const { scrollYProgress } = useScroll({
   178→   178→    offset: ['start start', 'end end'],
   179→   179→  });
   180→   180→
   181→   181→  const smoothProgress = useSpring(scrollYProgress, {
   182→   182→    stiffness: 100,
   183→   183→    damping: 30,
   184→   184→    restDelta: 0.001,
   185→   185→  });
   186→   186→
   187→   187→  useEffect(() => {
   188→   188→    const observers: IntersectionObserver[] = [];
   189→   189→
   190→   190→    chapters.forEach((chapter) => {
   191→   191→      const element = document.getElementById(chapter.id);
   192→   192→      if (!element) return;
   193→   193→
   194→   194→      const observer = new IntersectionObserver(
   195→   195→        ([entry]) => {
   196→   196→          if (entry.isIntersecting) {
   197→   197→            setActiveChapter(chapters.indexOf(chapter));
   198→   198→          }
   199→   199→        },
   200→   200→        { threshold: 0.5, rootMargin: '-20% 0px -20% 0px' }
   201→   201→      );
   202→   202→
   203→   203→      observer.observe(element);
   204→   204→      observers.push(observer);
   205→   205→    });
   206→   206→
   207→   207→    return () => observers.forEach((obs) => obs.disconnect());
   208→   208→  }, []);
   209→   209→
   210→   210→  const scrollToChapter = (index: number) => {
   211→   211→    const element = document.getElementById(chapters[index].id);
   212→   212→    element?.scrollIntoView({ behavior: 'smooth' });
   213→   213→  };
   214→   214→
   215→   215→  return (
   216→   216→    <motion.nav
   217→   217→      ref={navRef}
   218→   218→      initial={{ x: 50, opacity: 0 }}
   219→   219→      animate={{ x: 0, opacity: 1 }}
   220→   220→      transition={{ delay: 1, duration: 0.5 }}
   221→   221→      className='fixed right-6 top-1/2 -translate-y-1/2 z-40 hidden xl:flex flex-col items-center gap-4'
   222→   222→    >
   223→   223→      {/* Progress line */}
   224→   224→      <div className='absolute top-0 bottom-0 w-0.5 bg-sand/30 -z-10'>
   225→   225→        <motion.div
   226→   226→          className='w-full bg-bordeaux'
   227→   227→          style={{ height: smoothProgress }}
   228→   228→        />
   229→   229→      </div>
   230→   230→
   231→   231→      {chapters.map((chapter, index) => (
   232→   232→        <button
   233→   233→          key={chapter.id}
   234→   234→          onClick={() => scrollToChapter(index)}
   235→   235→          className={`group relative flex items-center justify-center w-8 h-8 rounded-full transition-all duration-300 ${
   236→   236→            activeChapter === index
   237→   237→              ? 'bg-bordeaux text-white'
   238→   238→              : 'bg-transparent text-muted hover:text-foreground'
   239→   239→          }`}
   240→   240→          aria-label={`Перейти к разделу ${chapter.label}`}
   241→   241→        >
   242→   242→          <span className='text-xs font-mono'>{String(index + 1).padStart(2, '0')}</span>
   243→   243→          
   244→   244→          {/* Tooltip on hover */}
   245→   245→          <span className='absolute right-full mr-4 px-3 py-1.5 bg-charcoal text-cream text-xs rounded opacity-0 group-hover:opacity-100 transition-opacity whitespace-nowrap pointer-events-none'>
   246→   246→            {chapter.label}
   247→   247→          </span>
   248→   248→        </button>
   249→   249→      ))}
   250→   250→    </motion.nav>
   251→   251→  );
   252→   252→}
   253→   253→```

   257→   257→### 3. Interactive Gallery with Lightbox
   258→   258→
   259→   259→```tsx
   260→   260→// components/catering/events-gallery.tsx (enhanced)
   261→   261→'use client';
   262→   262→
   263→   263→import { useState, useCallback } from 'react';
   264→   264→import { motion, AnimatePresence } from 'framer-motion';
   265→   265→import Image from 'next/image';
   266→   266→
   267→   267→interface GalleryItem {
   268→   268→  src: string;
   269→   269→  alt: string;
   270→   270→  category: string;
   271→   271→  title?: string;
   272→   272→}
   273→   273→
   274→   274→export function EventsGallery({ items }: { items: GalleryItem[] }) {
   275→   275→  const [activeFilter, setActiveFilter] = useState<string>('all');
   276→   276→  const [lightboxIndex, setLightboxIndex] = useState<number | null>(null);
   277→   277→  const [hoveredIndex, setHoveredIndex] = useState<number | null>(null);
   278→   278→
   279→   279→  const categories = ['all', ...new Set(items.map((i) => i.category))];
   280→   280→
   281→   281→  const filteredItems =
   282→   282→    activeFilter === 'all'
   283→   283→      ? items
   284→   284→      : items.filter((i) => i.category === activeFilter);
   285→   285→
   286→   286→  const openLightbox = useCallback((index: number) => {
   287→   287→    setLightboxIndex(index);
   288→   288→    document.body.style.overflow = 'hidden';
   289→   289→  }, []);
   290→   290→
   291→   291→  const closeLightbox = useCallback(() => {
   292→   292→    setLightboxIndex(null);
   293→   293→    document.body.style.overflow = '';
   294→   294→  }, []);
   295→   295→
   296→   296→  const navigateLightbox = (direction: 'prev' | 'next') => {
   297→   297→    if (lightboxIndex === null) return;
   298→   298→    const newIndex =
   299→   299→      direction === 'prev'
   300→   300→        ? (lightboxIndex - 1 + filteredItems.length) % filteredItems.length
   301→   301→        : (lightboxIndex + 1) % filteredItems.length;
   302→   302→    setLightboxIndex(newIndex);
   303→   303→  };
   304→   304→
   305→   305→  return (
   306→   306→    <>
   307→   307→      {/* Filter tabs */}
   308→   308→      <div className='flex flex-wrap justify-center gap-3 mb-12'>
   309→   309→        {categories.map((category) => (
   310→   310→          <button
   311→   311→            key={category}
   312→   312→            onClick={() => setActiveFilter(category)}
   313→   313→            className={`px-5 py-2 rounded-full text-sm capitalize transition-all duration-300 ${
   314→   314→              activeFilter === category
   315→   315→                ? 'bg-bordeaux text-white'
   316→   316→                : 'bg-parchment text-espresso hover:bg-sand'
   317→   317→            }`}
   318→   318→          >
   319→   319→            {category === 'all' ? 'Все' : category}
   320→   320→          </button>
   321→   321→        ))}
   322→   322→      </div>
   323→   323→
   324→   324→      {/* Grid */}
   325→   325→      <motion.div
   326→   326→        layout
   327→   327→        className='grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4'
   328→   328→      >
   329→   329→        <AnimatePresence mode='popLayout'>
   330→   330→          {filteredItems.map((item, index) => (
   331→   331→            <motion.div
   332→   332→              key={item.src}
   333→   333→              layout
   334→   334→              initial={{ opacity: 0, scale: 0.9 }}
   335→   335→              animate={{ opacity: 1, scale: 1 }}
   336→   336→              exit={{ opacity: 0, scale: 0.9 }}
   337→   337→              transition={{ duration: 0.3 }}
   338→   338→              className='group relative aspect-[4/3] overflow-hidden rounded-2xl cursor-pointer'
   339→   339→              onMouseEnter={() => setHoveredIndex(index)}
   340→   340→              onMouseLeave={() => setHoveredIndex(null)}
   341→   341→              onClick={() => openLightbox(index)}
   342→   342→            >
   343→   343→              <Image
   344→   344→                src={item.src}
   345→   345→                alt={item.alt}
   346→   346→                fill
   347→   347→                className='object-cover transition-transform duration-700 group-hover:scale-110'
   348→   348→              />
   349→   349→
   350→   350→              {/* Overlay */}
   351→   351→              <div
   352→   352→                className={`absolute inset-0 bg-gradient-to-t from-night via-night/40 to-transparent 
   353→   353→                  flex items-end p-6 transition-opacity duration-300 ${
   354→   354→                    hoveredIndex === index ? 'opacity-100' : 'opacity-0'
   355→   355→                  }`}
   356→   356→              >
   357→   357→                <div>
   358→   358→                  {item.title && (
   359→   359→                    <h3 className='text-white font-display text-xl'>{item.title}</h3>
   360→   360→                  )}
   361→   361→                  <p className='text-cream/70 text-sm capitalize'>{item.category}</p>
   362→   362→                </div>
   363→   363→              </div>
   364→   364→
   365→   365→              {/* Zoom icon */}
   366→   366→              <div
   367→   367→                className={`absolute top-4 right-4 w-10 h-10 rounded-full bg-white/20 backdrop-blur-sm
   368→   368→                  flex items-center justify-center transition-all duration-300 ${
   369→   369→                    hoveredIndex === index
   370→   370→                      ? 'opacity-100 translate-y-0'
   371→   371→                      : 'opacity-0 -translate-y-2'
   372→   372→                  }`}
   373→   373→              >
   374→   374→                <svg className='w-5 h-5 text-white' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   375→   375→                  <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0zM10 7v6m3-3H7' />
   376→   376→                </svg>
   377→   377→              </div>
   378→   378→            </motion.div>
   379→   379→          ))}
   380→   380→        </AnimatePresence>
   381→   381→      </motion.div>
   382→   382→
   383→   383→      {/* Lightbox */}
   384→   384→      <AnimatePresence>
   385→   385→        {lightboxIndex !== null && (
   386→   386→          <motion.div
   387→   387→            initial={{ opacity: 0 }}
   388→   388→            animate={{ opacity: 1 }}
   389→   389→            exit={{ opacity: 0 }}
   390→   390→            className='fixed inset-0 z-[100] bg-night/95 flex items-center justify-center p-4'
   391→   391→            onClick={closeLightbox}
   392→   392→          >
   393→   393→            {/* Close button */}
   394→   394→            <button
   395→   395→              onClick={closeLightbox}
   396→   396→              className='absolute top-6 right-6 w-12 h-12 rounded-full bg-white/10 flex items-center justify-center text-white hover:bg-white/20 transition-colors z-10'
   397→   397→              aria-label='Закрыть'
   398→   398→            >
   399→   399→              <svg className='w-6 h-6' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   400→   400→                <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M6 18L18 6M6 6l12 12' />
   401→   401→              </svg>
   402→   402→            </button>
   403→   403→
   404→   404→            {/* Image */}
   405→   405→            <motion.div
   406→   406→              key={lightboxIndex}
   407→   407→              initial={{ opacity: 0, scale: 0.9 }}
   408→   408→              animate={{ opacity: 1, scale: 1 }}
   409→   409→              exit={{ opacity: 0, scale: 0.9 }}
   410→   410→              className='relative max-w-5xl max-h-[80vh] aspect-video'
   411→   411→              onClick={(e) => e.stopPropagation()}
   412→   412→            >
   413→   413→              <Image
   414→   414→                src={filteredItems[lightboxIndex].src}
   415→   415→                alt={filteredItems[lightboxIndex].alt}
   416→   416→                fill
   417→   417→                className='object-contain'
   418→   418→              />
   419→   419→
   420→   420→              {/* Caption */}
   421→   421→              <div className='absolute bottom-0 left-0 right-0 p-6 bg-gradient-to-t from-night to-transparent'>
   422→   422→                <h3 className='text-white font-display text-xl'>
   423→   423→                  {filteredItems[lightboxIndex].title || filteredItems[lightboxIndex].alt}
   424→   424→                </h3>
   425→   425→              </div>
   426→   426→            </motion.div>
   427→   427→
   428→   428→            {/* Navigation arrows */}
   429→   429→            <button
   430→   430→              onClick={(e) => {
   431→   431→                e.stopPropagation();
   432→   432→                navigateLightbox('prev');
   433→   433→              }}
   434→   434→              className='absolute left-4 top-1/2 -translate-y-1/2 w-12 h-12 rounded-full bg-white/10 flex items-center justify-center text-white hover:bg-white/20 transition-colors'
   435→   435→              aria-label='Предыдущее изображение'
   436→   436→            >
   437→   437→              <svg className='w-6 h-6' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   438→   438→                <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M15 19l-7-7 7-7' />
   439→   439→              </svg>
   440→   440→            </button>
   441→   441→            <button
   442→   442→              onClick={(e) => {
   443→   443→                e.stopPropagation();
   444→   444→                navigateLightbox('next');
   445→   445→              }}
   446→   446→              className='absolute right-4 top-1/2 -translate-y-1/2 w-12 h-12 rounded-full bg-white/10 flex items-center justify-center text-white hover:bg-white/20 transition-colors'
   447→   447→              aria-label='Следующее изображение'
   448→   448→            >
   449→   449→              <svg className='w-6 h-6' fill='none' viewBox='0 0 24 24' stroke='currentColor'>
   450→   450→                <path strokeLinecap='round' strokeLinejoin='round' strokeWidth={2} d='M9 5l7 7-7 7' />
   451→   451→              </svg>
   452→   452→            </button>
   453→   453→
   454→   454→            {/* Counter */}
   455→   455→            <div className='absolute bottom-6 left-1/2 -translate-x-1/2 text-white/60 text-sm'>
   456→   456→              {lightboxIndex + 1} / {filteredItems.length}
   457→   457→            </div>
   458→   458→          </motion.div>
   459→   459→        )}
   460→   460→      </AnimatePresence>
   461→   461→    </>
   462→   462→  );
   463→   463→}
   464→   464→```

   466→   466→### 4. Testimonial Carousel (Ridgewells style)
   467→   467→
   468→   468→```tsx
   469→   469→// components/catering/testimonials.tsx (enhanced)
   470→   470→'use client';
   471→   471→

   476→   476→interface Testimonial {
   477→   477→  company: string;

   479→   479→  author: string;
   480→   480→  role: string;
   481→   481→  rating: number;

   483→   483→}
   484→   484→
   485→   485→export function TestimonialCarousel({ testimonials }: { testimonials: Testimonial[] }) {
   486→   486→  const [current, setCurrent] = useState(0);
   487→   487→  const [direction, setDirection] = useState(0);
   488→   488→
   489→   489→  const next = useCallback(() => {
   490→   490→    setDirection(1);
   491→   491→    setCurrent((prev) => (prev + 1) % testimonials.length);
   492→   492→  }, [testimonials.length]);
   493→   493→
   494→   494→  const prev = useCallback(() => {
   495→   495→    setDirection(-1);
   496→   496→    setCurrent((prev) => (prev - 1 + testimonials.length) % testimonials.length);
   497→   497→  }, [testimonials.length]);
   498→   498→
   499→   499→  // Auto-advance
   500→   500→  useEffect(() => {
   501→   501→    const timer = setInterval(next, 8000);
   502→   502→    return () => clearInterval(timer);
   503→   503→  }, [next]);
   504→   504→
   505→   505→  const variants = {
   506→   506→    enter: (direction: number) => ({
   507→   507→      x: direction > 0 ? 300 : -300,
   508→   508→      opacity: 0,
   509→   509→    }),
   510→   510→    center: {
   511→   511→      x: 0,
   512→   512→      opacity: 1,
   513→   513→    }),
   514→   514→    exit: (direction: number) => ({
   515→   515→      x: direction < 0 ? 300 : -300,
   516→   516→      opacity: 0,
   517→   517→    }),
   518→   518→  };
   519→   519→
   520→   520→  return (
   521→   521→    <div className='relative max-w-4xl mx-auto'>

   530→   530→          <motion.div
   531→   531→            key={current}
   532→   532→            custom={direction}
   533→   533→            variants={variants}
   534→   534→            initial='enter'

   536→   536→            exit='exit'

   540→   540→            }}
   541→   541→            className='w-full text-center px-8'
   542→   542→          >

   547→   547→                  key={i}

   549→   549→                    i < testimonials[current].rating
   550→   550→                      ? 'fill-gold text-gold'
   551→   551→                      : 'text-sand'
   552→   552→                  }`}
   553→   553→                />
   554→   554→              ))}

   565→   565→                <div className='w-14 h-14 rounded-full overflow-hidden'>

   575→   575→              <div className='text-left'>
   576→   576→                <cite className='not-italic font-semibold text-white'>
   577→   577→                  {testimonials[current].author}
   578→   578→                </cite>
   579→   579→                <p className='text-sm text-cream/60'>

   582→   582→              </div>

   584→   584→          </motion.div>

   586→   586→      </div>
   587→   587→
   588→   588→      {/* Navigation */}
   589→   589→      <div className='flex items-center justify-center gap-6 mt-8'>
   590→   590→        <button
   591→   591→          onClick={prev}
   592→   592→          className='w-12 h-12 rounded-full border border-sand/30 flex items-center justify-center text-cream/60 hover:text-bordeau hover:border-bordeaux transition-colors'

   812→   812→### 6. Accordion / FAQ Component
   813→   813→
   814→   814→```tsx
   815→   815→// components/ui/accordion.tsx (enhanced)
   816→   816→'use client';
   817→   817→

   874→   874→}

   904→   904→### 8. Back to Top Button
   905→   905→
   906→   906→```tsx
   907→   907→// components/motion/back-to-top.tsx
   908→   908→'use client';
   909→   909→
   910→   910→import { useState, useEffect } from 'react';
   911→   911→import { motion, AnimatePresence } from 'framer-motion';
   912→   912→import { ArrowUp } from 'lucide-react';
   913→   913→
   914→   914→export function BackToTop() {
   915→   915→  const [isVisible, setIsVisible] = useState(false);
   916→   916→
   917→   917→  useEffect(() => {
   918→   918→    const toggleVisibility = () => {
   919→   919→      setIsVisible(window.scrollY > 500);
   920→   920→    };
   921→   921→
   922→   922→    window.addEventListener('scroll', toggleVisibility, { passive: true });
   923→   923→    return () => window.removeEventListener('scroll', toggleVisibility);
   924→   924→  }, []);
   925→   925→
   926→   926→  const scrollToTop = () => {
   927→   927→    window.scrollTo({ top: 0, behavior: 'smooth' });
   928→   928→  };
   929→   929→
   930→   930→  return (
   931→   931→    <AnimatePresence>
   932→   932→      {isVisible && (
   933→   933→        <motion.button
   934→   934→          initial={{ opacity: 0, y: 20 }}
   935→   935→          animate={{ opacity: 1, y: 0 }}
   936→   936→          exit={{ opacity: 0, y: 20 }}
   937→   937→          onClick={scrollToTop}
   938→   938→          className='fixed bottom-8 right-8 z-50 w-12 h-12 rounded-full bg-bordeaux text-white
   939→   939→            flex items-center justify-center shadow-lg hover:bg-bordeaux-dark transition-colors'
   940→   940→          aria-label='Наверх'
   941→   941→        >
   942→   942→          <ArrowUp className='w-5 h-5' />
   943→   943→        </motion.button>
   944→   944→      )}
   945→   945→    </AnimatePresence>
   946→   946→  );
   947→   947→}
