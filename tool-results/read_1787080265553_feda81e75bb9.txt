    15→### 1. Hero Cinematic Entrance (Wolfgang Puck / Pinch style)
    16→
    17→```tsx
    18→// components/motion/hero-cinematic.tsx
    19→'use client';
    20→
    21→import { useRef, useEffect } from 'react';
    22→import { motion, useScroll, useTransform } from 'framer-motion';
    23→
    24→export function HeroCinematic({ children }: { children: React.ReactNode }) {
    25→  const containerRef = useRef<HTMLDivElement>(null);
    26→  const { scrollYProgress } = useScroll({
    27→    target: containerRef,
    28→    offset: ['start start', 'end end'],
    29→  });
    30→
    31→  // Parallax layers
    32→  const y = useTransform(scrollYProgress, [0, 1], ['0%', '30%']);
    33→  const opacity = useTransform(scrollYProgress, [0, 0.8], [1, 0]);
    34→  const scale = useTransform(scrollYProgress, [0, 0.8], [1, 0.95]);
    35→
    36→  return (
    37→    <motion.div
    38→      ref={containerRef}
    39→      style={{ y, opacity, scale }}
    40→      className="relative min-h-screen overflow-hidden"
    41→    >
    42→      {children}
    43→    </motion.div>
    44→  );
    45→}
    46→
    47→// Letter-by-letter reveal animation
    48→export function AnimatedText({
    49→  text,
    50→  className,
    51→}: {
    52→  text: string;
    53→  className?: string;
    54→}) {
    55→  const words = text.split(' ');
    56→
    57→  const container = {
    58→    hidden: { opacity: 0 },
    59→    visible: (i = 1) => ({
    60→      opacity: 1,
    61→      transition: { staggerChildren: 0.08, delayChildren: 0.3 * i },
    62→    }),
    63→  };
    64→
    65→  const child = {
    66→    visible: {
    67→      opacity: 1,
    68→      y: 0,
    69→      transition: {
    70→        type: 'spring',
    71→        damping: 12,
    72→        stiffness: 100,
    73→      },
    74→    },
    75→    hidden: {
    76→      opacity: 0,
    77→      y: 40,
    78→      transition: {
    79→        type: 'spring',
    80→        damping: 12,
    81→        stiffness: 100,
    82→      },
    83→    },
    84→  };
    85→
    86→  return (
    87→    <motion.div
    88→      className={`flex flex-wrap ${className ?? ''}`}
    89→      variants={container}
    90→      initial='hidden'
    91→      animate='visible'
    92→    >
    93→      {words.map((word, index) => (
    94→        <motion.span
    95→          key={index}
    96→          variants={child}
    97→          className='mr-2'
    98→        >
    99→          {word}
   100→        </motion.span>
   101→      ))}
   102→    </motion.div>
   103→  );
   104→}

   107→### 2. Magnetic Button Effect (Pinch "Book Now" style)
   108→
   109→```tsx
   110→// components/motion/magnetic-button.tsx
   111→'use client';
   112→
   113→import { useRef, useState } from 'react';
   114→import { motion, useMotionValue, useSpring, useTransform } from 'framer-motion';
   115→
   116→interface MagneticButtonProps {
   117→  children: React.ReactNode;
   118→  className?: string;
   119→  strength?: number;
   120→  onClick?: () => void;
   121→}
   122→
   123→export function MagneticButton({
   124→  children,
   125→  className,
   126→  strength = 40,
   127→  onClick,
   128→}: MagneticButtonProps) {
   129→  const ref = useRef<HTMLButtonElement>(null);
   130→  const [isHovered, setIsHovered] = useState(false);
   131→
   132→  const x = useMotionValue(0);
   133→  const y = useMotionValue(0);
   134→
   135→  const springX = useSpring(x, { stiffness: 300, damping: 20 });
   136→  const springY = useSpring(y, { stiffness: 300, damping: 20 });
   137→
   138→  const handleMouseMove = (e: React.MouseEvent) => {
   139→    if (!ref.current) return;
   140→
   141→    const rect = ref.current.getBoundingClientRect();
   142→    const centerX = rect.left + rect.width / 2;
   143→    const centerY = rect.top + rect.height / 2;
   144→
   145→    const deltaX = (e.clientX - centerX) / (rect.width / 2);
   146→    const deltaY = (e.clientY - centerY) / (rect.height / 2);
   147→
   148→    x.set(deltaX * strength);
   149→    y.set(deltaY * strength);
   150→  };
   151→
   152→  const handleMouseLeave = () => {
   153→    x.set(0);
   154→    y.set(0);
   155→    setIsHovered(false);
   156→  };
   157→
   158→  return (
   159→    <motion.button
   160→      ref={ref}
   161→      style={{ x: springX, y: springY }}
   162→      onMouseMove={handleMouseMove}
   163→      onMouseEnter={() => setIsHovered(true)}
   164→      onMouseLeave={handleMouseLeave}
   165→      onClick={onClick}
   166→      className={`relative ${className ?? ''}`}
   167→      whileTap={{ scale: 0.97 }}
   168→    >
   169→      {/* Glow effect on hover */}
   170→      <motion.div
   171→        className='absolute inset-0 rounded-inherit'
   172→        initial={{ opacity: 0 }}
   173→        animate={{
   174→          opacity: isHovered ? 1 : 0,
   175→          boxShadow: isHovered
   176→            ? '0 0 40px rgba(209, 26, 70, 0.3)'
   177→            : '0 0 0px transparent',
   178→        }}
   179→        transition={{ duration: 0.3 }}
   180→      />
   181→      {children}
   182→    </motion.button>
   183→  );
   184→}
   185→```

   187→### 3. Scroll-Triggered Reveals (GSAP ScrollTrigger style)
   188→
   189→```tsx
   190→// components/motion/scroll-reveal.tsx
   191→'use client';
   192→
   193→import { useRef, useEffect } from 'react';
   194→import { motion, useInView, useAnimation } from 'framer-motion';
   195→
   196→type RevealVariant = 'fade-up' | 'fade-down' | 'fade-left' | 'fade-right' | 'scale-up' | 'blur-in';
   197→
   198→interface ScrollRevealProps {
   199→  children: React.ReactNode;
   200→  variant?: RevealVariant;
   201→  delay?: number;
   202→  duration?: number;
   203→  className?: string;
   204→  once?: boolean;
   205→  threshold?: number;
   206→}
   207→
   208→const variantMap: Record<RevealVariant, { hidden: object; visible: object }> = {
   209→  'fade-up': {
   210→    hidden: { opacity: 0, y: 60 },
   211→    visible: { opacity: 1, y: 0 },
   212→  },
   213→  'fade-down': {
   214→    hidden: { opacity: 0, y: -60 },
   215→    visible: { opacity: 1, y: 0 },
   216→  },
   217→  'fade-left': {
   218→    hidden: { opacity: 0, x: -60 },
   219→    visible: { opacity: 1, x: 0 },
   220→  },
   221→  'fade-right': {
   222→    hidden: { opacity: 0, x: 60 },
   223→    visible: { opacity: 1, x: 0 },
   224→  },
   225→  'scale-up': {
   226→    hidden: { opacity: 0, scale: 0.85 },
   227→    visible: { opacity: 1, scale: 1 },
   228→  },
   229→  'blur-in': {
   230→    hidden: { opacity: 0, filter: 'blur(10px)' },
   231→    visible: { opacity: 1, filter: 'blur(0px)' },
   232→  },
   233→};
   234→
   235→export function ScrollReveal({
   236→  children,
   237→  variant = 'fade-up',
   238→  delay = 0,
   239→  duration = 0.6,
   240→  className,
   241→  once = true,
   242→  threshold = 0.2,
   243→}: ScrollRevealProps) {
   244→  const ref = useRef<HTMLDivElement>(null);
   245→  const isInView = useInView(ref, { once, amount: threshold });
   246→  const controls = useAnimation();
   247→
   248→  const selectedVariant = variantMap[variant];
   249→
   250→  useEffect(() => {
   251→    if (isInView) {
   252→      controls.start('visible');
   253→    }
   254→  }, [isInView, controls]);
   255→
   256→  return (
   257→    <motion.div
   258→      ref={ref}
   259→      initial='hidden'
   260→      animate={controls}
   261→      variants={selectedVariant}
   262→      transition={{
   263→        duration,
   264→        delay,
   265→        ease: [0.25, 0.4, 0.25, 1],
   266→      }}
   267→      className={className}
   268→    >
   269→      {children}
   270→    </motion.div>
   271→  );
   272→}
   273→
   274→// Staggered children reveal
   275→export function StaggerContainer({
   276→  children,
   277→  className,
   278→  staggerDelay = 0.1,
   279→}: {
   280→  children: React.ReactNode;
   281→  className?: string;
   282→  staggerDelay?: number;
   283→}) {
   284→  const ref = useRef<HTMLDivElement>(null);
   285→  const isInView = useInView(ref, { once: true, amount: 0.2 });
   286→
   287→  return (
   288→    <motion.div
   289→      ref={ref}
   290→      initial='hidden'
   291→      animate={isInView ? 'visible' : 'hidden'}
   292→      variants={{
   293→        hidden: {},
   294→        visible: {
   295→          transition: {
   296→            staggerChildren: staggerDelay,
   297→          },
   298→        },
   299→      }}
   300→      className={className}
   301→    >
   302→      {children}
   303→    </motion.div>
   304→  );
   305→}

   308→  return (

   312→        visible: {
   313→          opacity: 1,
   314→          y: 0,
   315→          transition: { duration: 0.5, ease: [0.25, 0.4, 0.25, 1] },
   316→        },
   317→      }}
   318→      className={className}
   319→    >
   320→      {children}
   321→    </motion.div>
   322→  );
   323→}

   326→### 4. Parallax Layers (M Culinary style)
   327→
   328→```tsx
   329→// components/motion/parallax-layers.tsx
   330→'use client';
   331→
   332→import { useRef } from 'react';
   333→import { motion, useScroll, useTransform } from 'framer-motion';
   334→
   335→interface ParallaxLayerProps {
   336→  children: React.ReactNode;
   337→  speed?: number; // 0 = static, 1 = normal, 2 = 2x speed
   338→  className?: string;
   339→  direction?: 'up' | 'down';
   340→}
   341→
   342→export function ParallaxLayer({
   343→  children,
   344→  speed = 0.5,
   345→  className,
   346→  direction = 'up',
   347→}: ParallaxLayerProps) {
   348→  const ref = useRef<HTMLDivElement>(null);
   349→
   350→  const { scrollYProgress } = useScroll({
   351→    target: ref,
   352→    offset: ['start end', 'end start'],
   353→  });
   354→
   355→  // Direction determines movement direction
   356→  const multiplier = direction === 'up' ? -1 : 1;
   357→  const range = 100 * speed * multiplier;
   358→
   359→  const y = useTransform(scrollYProgress, [0, 1], [range, -range]);
   360→
   361→  return (
   362→    <div ref={ref} className={`relative overflow-hidden ${className ?? ''}`}>
   363→      <motion.div style={{ y }} className='h-full w-full'>
   364→        {children}
   365→      </motion.div>
   366→    </div>
   367→  );
   368→}
   369→
   370→// Multi-layer parallax section
   371→interface ParallaxSectionProps {
   372→  layers: Array<{
   373→    component: React.ReactNode;
   374→    speed: number;
   375→    className?: string;
   376→  }>;
   377→  className?: string;
   378→  height?: string;
   379→}
   380→
   381→export function ParallaxSection({ layers, className, height = '100vh' }: ParallaxSectionProps) {
   382→  const containerRef = useRef<HTMLDivElement>(null);
   383→
   384→  return (
   385→    <section ref={containerRef} className={`relative ${className}`} style={{ height }}>
   386→      {layers.map((layer, index) => (
   387→        <ParallaxLayer key={index} speed={layer.speed} className={layer.className}>
   388→          {layer.component}
   389→        </ParallaxLayer>
   390→      ))}
   391→    </section>
   392→  );
   393→}
   394→```

   396→### 5. Animated Counters (Pinch Live Stats style)
   397→
   398→```tsx
   399→// components/motion/animated-counter.tsx
   400→'use client';
   401→
   402→import { useEffect, useState, useRef } from 'react';
   403→import { useInView, motion } from 'framer-motion';
   404→
   405→interface AnimatedCounterProps {
   406→  target: number;
   407→  duration?: number; // ms
   408→  prefix?: string;
   409→  suffix?: string;
   410→  decimals?: number;
   411→  className?: string;
   412→  label?: string;
   413→}
   414→
   415→export function AnimatedCounter({
   416→  target,
   417→  duration = 2000,
   418→  prefix = '',
   419→  suffix = '',
   420→  decimals = 0,
   421→  className,
   422→  label,
   423→}: AnimatedCounterProps) {
   424→  const [count, setCount] = useState(0);
   425→  const ref = useRef<HTMLSpanElement>(null);
   426→  const isInView = useInView(ref, { once: true });
   427→  const hasAnimated = useRef(false);
   428→
   429→  useEffect(() => {
   430→    if (!isInView || hasAnimated.current) return;
   431→    hasAnimated.current = true;
   432→
   433→    const startTime = Date.now();
   434→    const animate = () => {
   435→      const elapsed = Date.now() - startTime;
   436→      const progress = Math.min(elapsed / duration, 1);
   437→
   438→      // Easing function (ease-out-expo)
   439→      const eased = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
   440→
   441→      setCount(eased * target);
   442→
   443→      if (progress < 1) {
   444→        requestAnimationFrame(animate);
   445→      }
   446→    };
   447→
   448→    requestAnimationFrame(animate);
   449→  }, [isInView, target, duration]);
   450→
   451→  const formatNumber = (num: number) => {
   452→    return num.toFixed(decimals).replace(/\B(?=(\d{3})+(?!\d))/g, ' ');
   453→  };
   454→
   455→  return (
   456→    <div className={`text-center ${className ?? ''}`}>
   457→      <motion.span
   458→        ref={ref}
   459→        className='block text-4xl md:text-5xl lg:text-6xl font-bold tracking-tight'
   460→        initial={{ opacity: 0, scale: 0.5 }}
   461→        animate={isInView ? { opacity: 1, scale: 1 } : {}}
   462→        transition={{ duration: 0.5, ease: 'easeOut' }}
   463→      >
   464→        {prefix}{formatNumber(count)}{suffix}
   465→      </motion.span>
   466→      {label && (
   467→        <span className='mt-2 block text-sm uppercase tracking-widest text-muted'>
   468→          {label}
   469→        </span>
   470→      )}
   471→    </div>
   472→  );
   473→}
   474→
   475→// Stats section component
   476→interface StatItem {
   477→  value: number;
   478→  prefix?: string;
   479→  suffix?: string;
   480→  label: string;
   481→}
   482→
   483→export function AnimatedStats({ stats }: { stats: StatItem[] }) {
   484→  return (
   485→    <div className='grid grid-cols-2 md:grid-cols-4 gap-8 lg:gap-16'>
   486→      {stats.map((stat, index) => (
   487→        <AnimatedCounter
   488→          key={index}
   489→          target={stat.value}
   490→          prefix={stat.prefix}
   491→          suffix={stat.suffix}
   492→          label={stat.label}
   493→          delay={index * 200}
   494→        />
   495→      ))}
   496→    </div>
   497→  );
   498→}
   499→```

   501→### 6. Custom Cursor (Premium trend)
   502→
   503→```tsx
   504→// components/motion/custom-cursor.tsx
   505→'use client';
   506→
   507→import { useEffect, useState } from 'react';
   508→import { motion, useMotionValue, useSpring } from 'framer-motion';
   509→
   510→export function CustomCursor() {
   511→  const [isVisible, setIsVisible] = useState(false);
   512→  const [isPointer, setIsPointer] = useState(false);
   513→  const [isHovering, setIsHovering] = useState(false);
   514→
   515→  const cursorX = useMotionValue(-100);
   516→  const cursorY = useMotionValue(-100);
   517→
   518→  const springConfig = { damping: 25, stiffness: 250 };
   519→  const cursorXSpring = useSpring(cursorX, springConfig);
   520→  const cursorYSpring = useSpring(cursorY, springConfig);
   521→
   522→  // Trailing ring with more lag
   523→  const ringX = useSpring(cursorX, { damping: 20, stiffness: 100 });
   524→  const ringY = useSpring(cursorY, { damping: 20, stiffness: 100 });
   525→
   526→  useEffect(() => {
   527→    // Only show custom cursor on desktop
   528→    if (window.matchMedia('(pointer: coarse)').matches) return;
   529→
   530→    const moveCursor = (e: MouseEvent) => {
   531→      cursorX.set(e.clientX);
   532→      cursorY.set(e.clientY);
   533→      if (!isVisible) setIsVisible(true);
   534→    };
   535→
   536→    const handleMouseOver = (e: MouseEvent) => {
   537→      const target = e.target as HTMLElement;
   538→      if (
   539→        target.tagName === 'A' ||
   540→        target.tagName === 'BUTTON' ||
   541→        target.closest('a') ||
   542→        target.closest('button') ||
   543→        target.dataset.cursor === 'pointer'
   544→      ) {
   545→        setIsPointer(true);
   546→      }
   547→
   548→      if (target.dataset.cursor === 'hover') {
   549→        setIsHovering(true);
   550→      }
   551→    };
   552→
   553→    const handleMouseOut = () => {
   554→      setIsPointer(false);
   555→      setIsHovering(false);
   556→    };
   557→
   558→    const handleMouseLeave = () => {
   559→      setIsVisible(false);
   560→    };
   561→
   562→    window.addEventListener('mousemove', moveCursor);
   563→    window.addEventListener('mouseover', handleMouseOver);
   564→    window.addEventListener('mouseout', handleMouseOut);
   565→    document.documentElement.addEventListener('mouseleave', handleMouseLeave);
   566→
   567→    return () => {
   568→      window.removeEventListener('mousemove', moveCursor);
   569→      window.removeEventListener('mouseover', handleMouseOver);
   570→      window.removeEventListener('mouseout', handleMouseOut);
   571→      document.documentElement.removeEventListener('mouseleave', handleMouseLeave);
   572→    };
   573→  }, []);
   574→
   575→  if (!isVisible) return null;
   576→
   577→  return (
   578→    <>
   579→      {/* Main dot */}
   580→      <motion.div
   581→        className='fixed top-0 left-0 pointer-events-none z-[9999] mix-blend-difference'
   582→        style={{
   583→          x: cursorXSpring,
   584→          y: cursorYSpring,
   585→        }}
   586→      >
   587→        <motion.div
   588→          className='rounded-full bg-white'
   589→          animate={{
   590→            width: isPointer ? 48 : isHovering ? 64 : 12,
   591→            height: isPointer ? 48 : isHovering ? 64 : 12,
   592→          }}
   593→          transition={{ type: 'spring', stiffness: 300, damping: 20 }}
   594→          style={{
   595→            marginLeft: isPointer ? -24 : isHovering ? -32 : -6,
   596→            marginTop: isPointer ? -24 : isHovering ? -32 : -6,
   597→          }}
   598→        />
   599→      </motion.div>
   600→
   601→      {/* Trailing ring */}
   602→      <motion.div
   603→        className='fixed top-0 left-0 pointer-events-none z-[9998]'
   604→        style={{
   605→          x: ringX,
   606→          y: ringY,
   607→        }}
   608→      >
   609→        <motion.div
   610→          className='rounded-full border border-current'
   611→          animate={{
   612→            width: isHovering ? 80 : 40,
   613→            height: isHovering ? 80 : 40,
   614→            opacity: isHovering ? 0.5 : 0.3,
   615→          }}
   616→          transition={{ type: 'spring', stiffness: 150, damping: 15 }}
   617→          style={{
   618→            marginLeft: isHovering ? -40 : -20,
   619→            marginTop: isHovering ? -40 : -20,
   620→          }}
   621→        />
   622→      </motion.div>
   623→    </>
   624→  );
   625→}
   626→```

   687→### 8. Text Scramble Effect (Cyberpunk/Premium style)
   688→
   689→```tsx
   690→// components/motion/text-scramble.tsx
   691→'use client';
   692→
   693→import { useEffect, useState, useRef } from 'react';
   694→
   695→const CHARS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%^&*';
   696→
   697→export function TextScramble({
   698→  text,
   699→  className,
   700→  trigger = true,
   701→}: {
   702→  text: string;
   703→  className?: string;
   704→  trigger?: boolean;
   705→}) {
   706→  const [displayText, setDisplayText] = useState(text);
   707→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   708→  const frameRef = useRef(0);
   709→  const textRef = useRef(text);
   710→
   711→  useEffect(() => {
   712→    textRef.current = text;
   713→  }, [text]);
   714→
   715→  useEffect(() => {
   716→    if (!trigger) {
   717→      setDisplayText(textRef.current);
   718→      return;
   719→    }
   720→
   721→    let iteration = 0;
   722→    const totalIterations = 15;
   723→
   724→    clearInterval(intervalRef.current!);
   725→
   726→    intervalRef.current = setInterval(() => {
   727→      setDisplayText(
   728→        textRef.current
   729→          .split('')
   730→          .map((char, index) => {
   731→            if (char === ' ') return ' ';
   732→            if (iteration > index + 5) return textRef.current[index];
   733→            return CHARS[Math.floor(Math.random() * CHARS.length)];
   734→          })
   735→          .join('')
   736→      );
   737→
   738→      iteration++;
   739→
   740→      if (iteration >= totalIterations + textRef.current.length) {
   741→        clearInterval(intervalRef.current!);
   742→        setDisplayText(textRef.current);
   743→      }
   744→    }, 35);
   745→
   746→    return () => {
   747→      clearInterval(intervalRef.current!);
   748→    };
   749→  }, [trigger]);
   750→
   751→  return <span className={className}>{displayText}</span>;
   752→}
   753→```

   758→// lib/gsap-utils.ts
   759→import { gsap } from 'gsap';

   766→// Pin and parallax section (Manifesto-style)
   767→export function createPinnedSection(
   768→  selector: string,
   769→  options?: {
   770→    start?: string;
   771→    end?: string;
   772→    scrub?: boolean | number;
   773→    pinSpacing?: boolean;
   774→  }
   775→) {
   776→  gsap.to(selector, {
   777→    scrollTrigger: {
   778→      trigger: selector,
   779→      start: options?.start ?? 'top top',
   780→      end: options?.end ?? 'bottom top',
   781→      pin: true,
   782→      scrub: options?.scrub ?? 1,
   783→      pinSpacing: options?.pinSpacing ?? false,
   784→    },
   785→  });
   786→}
   787→
   788→// Text color progression on scroll
   789→export function createTextColorReveal(
   790→  wordsSelector: string,
   791→  options?: {
   792→    trigger?: string;
   793→    start?: string;
   794→    end?: string;
   795→    scrub?: boolean | number;
   796→  }
   797→) {
   798→  const words = document.querySelectorAll(wordsSelector);
   799→
   800→  words.forEach((word, index) => {
   801→    gsap.fromTo(
   802→      word,
   803→      { color: 'rgba(255, 255, 255, 0.14)' },
   804→      {
   805→        color: 'rgba(252, 251, 248, 1)',
   806→        scrollTrigger: {
   807→          trigger: options?.trigger ?? wordsSelector,
   808→          start: options?.start ?? 'top 80%',
   809→          end: options?.end ?? 'bottom 20%',
   810→          scrub: options?.scrub ?? true,
   811→          containerAnimation: gsap.to(wordsSelector, {}),
   812→        },
   813→      }
   814→    );
   815→  });
   816→}
   817→
   818→// Horizontal scroll section
   819→export function createHorizontalScroll(
   820→  containerSelector: string,
   821→  itemsSelector: string
   822→) {
   823→  const container = document.querySelector(containerSelector);
   824→  const items = document.querySelectorAll(itemsSelector);
   825→
   826→  if (!container || !items.length) return;
   827→
   828→  const scrollWidth = items[items.length - 1].getBoundingClientRect().right -
   829→    container.getBoundingClientRect().left;
   830→
   831→  gsap.to(items, {
   832→    x: () => -(scrollWidth - window.innerWidth),
   833→    ease: 'none',
   834→    scrollTrigger: {
   835→      trigger: containerSelector,
   836→      start: 'top top',
   837→      end: () => `+=${scrollWidth}`,
   838→      scrub: 1,
   839→      pin: true,
   840→      anticipatePin: 1,
   841→    },
   842→  });
   843→}

  1017→## Performance Guidelines
  1018→
  1019→1. **GPU-accelerated properties only**: `transform`, `opacity`
  1020→2. **Use `will-change` sparingly** — only on actively animating elements
  1021→3. **Batch animations** — use `layoutAnimation` for layout changes
  1022→4. **Lazy mount** — dynamic import heavy animation components
  1023→5. **Debounce scroll handlers** — use `requestAnimationFrame`
  1024→6. **Test on mobile** — check performance on real devices
