    15→    15→### 1. Hero Cinematic Entrance (Wolfgang Puck / Pinch style)
    16→    16→
    17→    17→```tsx
    18→    18→// components/motion/hero-cinematic.tsx
    19→    19→'use client';
    20→    20→
    21→    21→import { useRef, useEffect } from 'react';
    22→    22→import { motion, useScroll, useTransform } from 'framer-motion';
    23→    23→
    24→    24→export function HeroCinematic({ children }: { children: React.ReactNode }) {
    25→    25→  const containerRef = useRef<HTMLDivElement>(null);
    26→    26→  const { scrollYProgress } = useScroll({
    27→    27→    target: containerRef,
    28→    28→    offset: ['start start', 'end end'],
    29→    29→  });
    30→    30→
    31→    31→  // Parallax layers
    32→    32→  const y = useTransform(scrollYProgress, [0, 1], ['0%', '30%']);
    33→    33→  const opacity = useTransform(scrollYProgress, [0, 0.8], [1, 0]);
    34→    34→  const scale = useTransform(scrollYProgress, [0, 0.8], [1, 0.95]);
    35→    35→
    36→    36→  return (
    37→    37→    <motion.div
    38→    38→      ref={containerRef}
    39→    39→      style={{ y, opacity, scale }}
    40→    40→      className="relative min-h-screen overflow-hidden"
    41→    41→    >
    42→    42→      {children}
    43→    43→    </motion.div>
    44→    44→  );
    45→    45→}
    46→    46→

   107→   107→### 2. Magnetic Button Effect (Pinch "Book Now" style)
   108→   108→
   109→   109→```tsx
   110→   110→// components/motion/magnetic-button.tsx
   111→   111→'use client';
   112→   112→
   113→   113→import { useRef, useState } from 'react';
   114→   114→import { motion, useMotionValue, useSpring, useTransform } from 'framer-motion';
   115→   115→
   116→   116→interface MagneticButtonProps {
   117→   117→  children: React.ReactNode;
   118→   118→  className?: string;
   119→   119→  strength?: number;
   120→   120→  onClick?: () => void;
   121→   121→}
   122→   122→
   123→   123→export function MagneticButton({
   124→   124→  children,
   125→   125→  className,
   126→   126→  strength = 40,
   127→   127→  onClick,
   128→   128→}: MagneticButtonProps) {
   129→   129→  const ref = useRef<HTMLButtonElement>(null);
   130→   130→  const [isHovered, setIsHovered] = useState(false);
   131→   131→
   132→   132→  const x = useMotionValue(0);
   133→   133→  const y = useMotionValue(0);
   134→   134→
   135→   135→  const springX = useSpring(x, { stiffness: 300, damping: 20 });
   136→   136→  const springY = useSpring(y, { stiffness: 300, damping: 20 });
   137→   137→
   138→   138→  const handleMouseMove = (e: React.MouseEvent) => {
   139→   139→    if (!ref.current) return;
   140→   140→
   141→   141→    const rect = ref.current.getBoundingClientRect();
   142→   142→    const centerX = rect.left + rect.width / 2;
   143→   143→    const centerY = rect.top + rect.height / 2;
   144→   144→
   145→   145→    const deltaX = (e.clientX - centerX) / (rect.width / 2);
   146→   146→    const deltaY = (e.clientY - centerY) / (rect.height / 2);
   147→   147→
   148→   148→    x.set(deltaX * strength);
   149→   149→    y.set(deltaY * strength);
   150→   150→  };
   151→   151→
   152→   152→  const handleMouseLeave = () => {
   153→   153→    x.set(0);
   154→   154→    y.set(0);
   155→   155→    setIsHovered(false);
   156→   156→  };
   157→   157→
   158→   158→  return (
   159→   159→    <motion.button
   160→   160→      ref={ref}
   161→   161→      style={{ x: springX, y: springY }}
   162→   162→      onMouseMove={handleMouseMove}
   163→   163→      onMouseEnter={() => setIsHovered(true)}
   164→   164→      onMouseLeave={handleMouseLeave}
   165→   165→      onClick={onClick}
   166→   166→      className={`relative ${className ?? ''}`}
   167→   167→      whileTap={{ scale: 0.97 }}
   168→   168→    >
   169→   169→      {/* Glow effect on hover */}
   170→   170→      <motion.div
   171→   171→        className='absolute inset-0 rounded-inherit'
   172→   172→        initial={{ opacity: 0 }}
   173→   173→        animate={{
   174→   174→          opacity: isHovered ? 1 : 0,
   175→   175→          boxShadow: isHovered
   176→   176→            ? '0 0 40px rgba(209, 26, 70, 0.3)'
   177→   177→            : '0 0 0px transparent',
   178→   178→        }}
   179→   179→        transition={{ duration: 0.3 }}
   180→   180→      />
   181→   181→      {children}
   182→   182→    </motion.button>
   183→   183→  );
   184→   184→}
   185→   185→```

   187→   187→### 3. Scroll-Triggered Reveals (GSAP ScrollTrigger style)
   188→   188→
   189→   189→```tsx
   190→   190→// components/motion/scroll-reveal.tsx
   191→   191→'use client';
   192→   192→
   193→   193→import { useRef, useEffect } from 'react';
   194→   194→import { motion, useInView, useAnimation } from 'framer-motion';
   195→   195→
   196→   196→type RevealVariant = 'fade-up' | 'fade-down' | 'fade-left' | 'fade-right' | 'scale-up' | 'blur-in';
   197→   197→
   198→   198→interface ScrollRevealProps {
   199→   199→  children: React.ReactNode;
   200→   200→  variant?: RevealVariant;
   201→   201→  delay?: number;
   202→   202→  duration?: number;
   203→   203→  className?: string;
   204→   204→  once?: boolean;
   205→   205→  threshold?: number;
   206→   206→}
   207→   207→
   208→   208→const variantMap: Record<RevealVariant, { hidden: object; visible: object }> = {
   209→   209→  'fade-up': {
   210→   210→    hidden: { opacity: 0, y: 60 },
   211→   211→    visible: { opacity: 1, y: 0 },
   212→   212→  },
   213→   213→  'fade-down': {
   214→   214→    hidden: { opacity: 0, y: -60 },
   215→   215→    visible: { opacity: 1, y: 0 },
   216→   216→  },
   217→   217→  'fade-left': {
   218→   218→    hidden: { opacity: 0, x: -60 },
   219→   219→    visible: { opacity: 1, x: 0 },
   220→   220→  },
   221→   221→  'fade-right': {
   222→   222→    hidden: { opacity: 0, x: 60 },
   223→   223→    visible: { opacity: 1, x: 0 },
   224→   224→  },
   225→   225→  'scale-up': {
   226→   226→    hidden: { opacity: 0, scale: 0.85 },
   227→   227→    visible: { opacity: 1, scale: 1 },
   228→   228→  },
   229→   229→  'blur-in': {
   230→   230→    hidden: { opacity: 0, filter: 'blur(10px)' },
   231→   231→    visible: { opacity: 1, filter: 'blur(0px)' },
   232→   232→  },
   233→   233→};
   234→   234→
   235→   235→export function ScrollReveal({
   236→   236→  children,
   237→   237→  variant = 'fade-up',
   238→   238→  delay = 0,
   239→   239→  duration = 0.6,
   240→   240→  className,
   241→   241→  once = true,
   242→   242→  threshold = 0.2,
   243→   243→}: ScrollRevealProps) {
   244→   244→  const ref = useRef<HTMLDivElement>(null);
   245→   245→  const isInView = useInView(ref, { once, amount: threshold });
   246→   246→  const controls = useAnimation();
   247→   247→
   248→   248→  const selectedVariant = variantMap[variant];
   249→   249→
   250→   250→  useEffect(() => {
   251→   251→    if (isInView) {
   252→   252→      controls.start('visible');
   253→   253→    }
   254→   254→  }, [isInView, controls]);
   255→   255→
   256→   256→  return (
   257→   257→    <motion.div
   258→   258→      ref={ref}
   259→   259→      initial='hidden'
   260→   260→      animate={controls}
   261→   261→      variants={selectedVariant}
   262→   262→      transition={{
   263→   263→        duration,
   264→   264→        delay,
   265→   265→        ease: [0.25, 0.4, 0.25, 1],
   266→   266→      }}
   267→   267→      className={className}
   268→   268→    >
   269→   269→      {children}
   270→   270→    </motion.div>
   271→   271→  );
   272→   272→}
   273→   273→

   326→   326→### 4. Parallax Layers (M Culinary style)
   327→   327→
   328→   328→```tsx
   329→   329→// components/motion/parallax-layers.tsx
   330→   330→'use client';
   331→   331→
   332→   332→import { useRef } from 'react';
   333→   333→import { motion, useScroll, useTransform } from 'framer-motion';
   334→   334→
   335→   335→interface ParallaxLayerProps {
   336→   336→  children: React.ReactNode;
   337→   337→  speed?: number; // 0 = static, 1 = normal, 2 = 2x speed
   338→   338→  className?: string;
   339→   339→  direction?: 'up' | 'down';
   340→   340→}
   341→   341→
   342→   342→export function ParallaxLayer({
   343→   343→  children,
   344→   344→  speed = 0.5,
   345→   345→  className,
   346→   346→  direction = 'up',
   347→   347→}: ParallaxLayerProps) {
   348→   348→  const ref = useRef<HTMLDivElement>(null);
   349→   349→
   350→   350→  const { scrollYProgress } = useScroll({
   351→   351→    target: ref,
   352→   352→    offset: ['start end', 'end start'],
   353→   353→  });
   354→   354→
   355→   355→  // Direction determines movement direction
   356→   356→  const multiplier = direction === 'up' ? -1 : 1;
   357→   357→  const range = 100 * speed * multiplier;
   358→   358→
   359→   359→  const y = useTransform(scrollYProgress, [0, 1], [range, -range]);
   360→   360→
   361→   361→  return (
   362→   362→    <div ref={ref} className={`relative overflow-hidden ${className ?? ''}`}>
   363→   363→      <motion.div style={{ y }} className='h-full w-full'>
   364→   364→        {children}
   365→   365→      </motion.div>
   366→   366→    </div>
   367→   367→  );
   368→   368→}
   369→   369→
   370→   370→// Multi-layer parallax section
   371→   371→interface ParallaxSectionProps {
   372→   372→  layers: Array<{
   373→   373→    component: React.ReactNode;
   374→   374→    speed: number;
   375→   375→    className?: string;
   376→   376→  }>;
   377→   377→  className?: string;
   378→   378→  height?: string;
   379→   379→}
   380→   380→
   381→   381→export function ParallaxSection({ layers, className, height = '100vh' }: ParallaxSectionProps) {
   382→   382→  const containerRef = useRef<HTMLDivElement>(null);
   383→   383→
   384→   384→  return (
   385→   385→    <section ref={containerRef} className={`relative ${className}`} style={{ height }}>
   386→   386→      {layers.map((layer, index) => (
   387→   387→        <ParallaxLayer key={index} speed={layer.speed} className={layer.className}>
   388→   388→          {layer.component}
   389→   389→        </ParallaxLayer>
   390→   390→      ))}
   391→   391→    </section>
   392→   392→  );
   393→   393→}
   394→   394→```

   396→   396→### 5. Animated Counters (Pinch Live Stats style)
   397→   397→
   398→   398→```tsx
   399→   399→// components/motion/animated-counter.tsx
   400→   400→'use client';
   401→   401→
   402→   402→import { useEffect, useState, useRef } from 'react';
   403→   403→import { useInView, motion } from 'framer-motion';
   404→   404→
   405→   405→interface AnimatedCounterProps {
   406→   406→  target: number;
   407→   407→  duration?: number; // ms
   408→   408→  prefix?: string;
   409→   409→  suffix?: string;
   410→   410→  decimals?: number;
   411→   411→  className?: string;
   412→   412→  label?: string;
   413→   413→}
   414→   414→
   415→   415→export function AnimatedCounter({
   416→   416→  target,
   417→   417→  duration = 2000,
   418→   418→  prefix = '',
   419→   419→  suffix = '',
   420→   420→  decimals = 0,
   421→   421→  className,
   422→   422→  label,
   423→   423→}: AnimatedCounterProps) {
   424→   424→  const [count, setCount] = useState(0);
   425→   425→  const ref = useRef<HTMLSpanElement>(null);
   426→   426→  const isInView = useInView(ref, { once: true });
   427→   427→  const hasAnimated = useRef(false);
   428→   428→
   429→   429→  useEffect(() => {
   430→   430→    if (!isInView || hasAnimated.current) return;
   431→   431→    hasAnimated.current = true;
   432→   432→
   433→   433→    const startTime = Date.now();
   434→   434→    const animate = () => {
   435→   435→      const elapsed = Date.now() - startTime;
   436→   436→      const progress = Math.min(elapsed / duration, 1);
   437→   437→
   438→   438→      // Easing function (ease-out-expo)
   439→   439→      const eased = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
   440→   440→
   441→   441→      setCount(eased * target);
   442→   442→
   443→   443→      if (progress < 1) {
   444→   444→        requestAnimationFrame(animate);
   445→   445→      }
   446→   446→    };
   447→   447→
   448→   448→    requestAnimationFrame(animate);
   449→   449→  }, [isInView, target, duration]);
   450→   450→
   451→   451→  const formatNumber = (num: number) => {
   452→   452→    return num.toFixed(decimals).replace(/\B(?=(\d{3})+(?!\d))/g, ' ');
   453→   453→  };
   454→   454→
   455→   455→  return (
   456→   456→    <div className={`text-center ${className ?? ''}`}>
   457→   457→      <motion.span
   458→   458→        ref={ref}
   459→   459→        className='block text-4xl md:text-5xl lg:text-6xl font-bold tracking-tight'
   460→   460→        initial={{ opacity: 0, scale: 0.5 }}
   461→   461→        animate={isInView ? { opacity: 1, scale: 1 } : {}}
   462→   462→        transition={{ duration: 0.5, ease: 'easeOut' }}
   463→   463→      >
   464→   464→        {prefix}{formatNumber(count)}{suffix}
   465→   465→      </motion.span>
   466→   466→      {label && (
   467→   467→        <span className='mt-2 block text-sm uppercase tracking-widest text-muted'>
   468→   468→          {label}
   469→   469→        </span>
   470→   470→      )}
   471→   471→    </div>
   472→   472→  );
   473→   473→}
   474→   474→
   475→   475→// Stats section component
   476→   476→interface StatItem {
   477→   477→  value: number;
   478→   478→  prefix?: string;
   479→   479→  suffix?: string;
   480→   480→  label: string;
   481→   481→}
   482→   482→
   483→   483→export function AnimatedStats({ stats }: { stats: StatItem[] }) {
   484→   484→  return (
   485→   485→    <div className='grid grid-cols-2 md:grid-cols-4 gap-8 lg:gap-16'>
   486→   486→      {stats.map((stat, index) => (
   487→   487→        <AnimatedCounter
   488→   488→          key={index}
   489→   489→          target={stat.value}
   490→   490→          prefix={stat.prefix}
   491→   491→          suffix={stat.suffix}
   492→   492→          label={stat.label}
   493→   493→          delay={index * 200}
   494→   494→        />
   495→   495→      ))}
   496→   496→    </div>
   497→   497→  );
   498→   498→}
   499→   499→```

   501→   501→### 6. Custom Cursor (Premium trend)
   502→   502→
   503→   503→```tsx
   504→   504→// components/motion/custom-cursor.tsx
   505→   505→'use client';
   506→   506→
   507→   507→import { useEffect, useState } from 'react';
   508→   508→import { motion, useMotionValue, useSpring } from 'framer-motion';
   509→   509→
   510→   510→export function CustomCursor() {
   511→   511→  const [isVisible, setIsVisible] = useState(false);
   512→   512→  const [isPointer, setIsPointer] = useState(false);
   513→   513→  const [isHovering, setIsHovering] = useState(false);
   514→   514→
   515→   515→  const cursorX = useMotionValue(-100);
   516→   516→  const cursorY = useMotionValue(-100);
   517→   517→
   518→   518→  const springConfig = { damping: 25, stiffness: 250 };
   519→   519→  const cursorXSpring = useSpring(cursorX, springConfig);
   520→   520→  const cursorYSpring = useSpring(cursorY, springConfig);
   521→   521→
   522→   522→  // Trailing ring with more lag
   523→   523→  const ringX = useSpring(cursorX, { damping: 20, stiffness: 100 });
   524→   524→  const ringY = useSpring(cursorY, { damping: 20, stiffness: 100 });
   525→   525→
   526→   526→  useEffect(() => {
   527→   527→    // Only show custom cursor on desktop
   528→   528→    if (window.matchMedia('(pointer: coarse)').matches) return;
   529→   529→
   530→   530→    const moveCursor = (e: MouseEvent) => {
   531→   531→      cursorX.set(e.clientX);
   532→   532→      cursorY.set(e.clientY);
   533→   533→      if (!isVisible) setIsVisible(true);
   534→   534→    };
   535→   535→
   536→   536→    const handleMouseOver = (e: MouseEvent) => {
   537→   537→      const target = e.target as HTMLElement;
   538→   538→      if (
   539→   539→        target.tagName === 'A' ||
   540→   540→        target.tagName === 'BUTTON' ||
   541→   541→        target.closest('a') ||
   542→   542→        target.closest('button') ||
   543→   543→        target.dataset.cursor === 'pointer'
   544→   544→      ) {
   545→   545→        setIsPointer(true);
   546→   546→      }
   547→   547→
   548→   548→      if (target.dataset.cursor === 'hover') {
   549→   549→        setIsHovering(true);
   550→   550→      }
   551→   551→    };
   552→   552→
   553→   553→    const handleMouseOut = () => {
   554→   554→      setIsPointer(false);
   555→   555→      setIsHovering(false);
   556→   556→    };
   557→   557→
   558→   558→    const handleMouseLeave = () => {
   559→   559→      setIsVisible(false);
   560→   560→    };
   561→   561→
   562→   562→    window.addEventListener('mousemove', moveCursor);
   563→   563→    window.addEventListener('mouseover', handleMouseOver);
   564→   564→    window.addEventListener('mouseout', handleMouseOut);
   565→   565→    document.documentElement.addEventListener('mouseleave', handleMouseLeave);
   566→   566→
   567→   567→    return () => {
   568→   568→      window.removeEventListener('mousemove', moveCursor);
   569→   569→      window.removeEventListener('mouseover', handleMouseOver);
   570→   570→      window.removeEventListener('mouseout', handleMouseOut);
   571→   571→      document.documentElement.removeEventListener('mouseleave', handleMouseLeave);
   572→   572→    };
   573→   573→  }, []);
   574→   574→
   575→   575→  if (!isVisible) return null;
   576→   576→
   577→   577→  return (
   578→   578→    <>
   579→   579→      {/* Main dot */}
   580→   580→      <motion.div
   581→   581→        className='fixed top-0 left-0 pointer-events-none z-[9999] mix-blend-difference'
   582→   582→        style={{
   583→   583→          x: cursorXSpring,
   584→   584→          y: cursorYSpring,
   585→   585→        }}
   586→   586→      >
   587→   587→        <motion.div
   588→   588→          className='rounded-full bg-white'
   589→   589→          animate={{
   590→   590→            width: isPointer ? 48 : isHovering ? 64 : 12,
   591→   591→            height: isPointer ? 48 : isHovering ? 64 : 12,
   592→   592→          }}
   593→   593→          transition={{ type: 'spring', stiffness: 300, damping: 20 }}
   594→   594→          style={{
   595→   595→            marginLeft: isPointer ? -24 : isHovering ? -32 : -6,
   596→   596→            marginTop: isPointer ? -24 : isHovering ? -32 : -6,
   597→   597→          }}
   598→   598→        />
   599→   599→      </motion.div>
   600→   600→
   601→   601→      {/* Trailing ring */}
   602→   602→      <motion.div
   603→   603→        className='fixed top-0 left-0 pointer-events-none z-[9998]'
   604→   604→        style={{
   605→   605→          x: ringX,
   606→   606→          y: ringY,
   607→   607→        }}
   608→   608→      >
   609→   609→        <motion.div
   610→   610→          className='rounded-full border border-current'
   611→   611→          animate={{
   612→   612→            width: isHovering ? 80 : 40,
   613→   613→            height: isHovering ? 80 : 40,
   614→   614→            opacity: isHovering ? 0.5 : 0.3,
   615→   615→          }}
   616→   616→          transition={{ type: 'spring', stiffness: 150, damping: 15 }}
   617→   617→          style={{
   618→   618→            marginLeft: isHovering ? -40 : -20,
   619→   619→            marginTop: isHovering ? -40 : -20,
   620→   620→          }}
   621→   621→        />
   622→   622→      </motion.div>

   687→   687→### 8. Text Scramble Effect (Cyberpunk/Premium style)
   688→   688→
   689→   689→```tsx
   690→   690→// components/motion/text-scramble.tsx
   691→   691→'use client';
   692→   692→
   693→   693→import { useEffect, useState, useRef } from 'react';
   694→   694→
   695→   695→const CHARS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%^&*';
   696→   696→
   697→   697→export function TextScramble({
   698→   698→  text,
   699→   699→  className,
   700→   700→  trigger = true,
   701→   701→}: {
   702→   702→  text: string;
   703→   703→  className?: string;
   704→   704→  trigger?: boolean;
   705→   705→}) {
   706→   706→  const [displayText, setDisplayText] = useState(text);
   707→   707→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   708→   708→  const frameRef = useRef(0);
   709→   709→  const textRef = useRef(text);
   710→   710→
   711→   711→  useEffect(() => {
   712→   712→    textRef.current = text;
   713→   713→  }, [text]);
   714→   714→
   715→   715→  useEffect(() => {
   716→   716→    if (!trigger) {
   717→   717→      setDisplayText(textRef.current);
   718→   718→      return;
   719→   719→    }
   720→   720→
   721→   721→    let iteration = 0;
   722→   722→    const totalIterations = 15;
   723→   723→
   724→   724→    clearInterval(intervalRef.current!);
   725→   725→
   726→   726→    intervalRef.current = setInterval(() => {
   727→   727→      setDisplayText(
   728→   728→        textRef.current
   729→   729→          .split('')
   730→   730→          .map((char, index) => {
   731→   731→            if (char === ' ') return ' ';
   732→   732→            if (iteration > index + 5) return textRef.current[index];
   733→   733→            return CHARS[Math.floor(Math.random() * CHARS.length)];
   734→   734→          })
   735→   735→          .join('')
   736→   736→      );
   737→   737→
   738→   738→      iteration++;
   739→   739→
   740→   740→      if (iteration >= totalIterations + textRef.current.length) {
   741→   741→        clearInterval(intervalRef.current!);
   742→   742→        setDisplayText(textRef.current);
   743→   743→      }
   744→   744→    }, 35);
   745→   745→
   746→   746→    return () => {
   747→   747→      clearInterval(intervalRef.current!);
   748→   748→    };
   749→   749→  }, [trigger]);
   750→   750→
   751→   751→  return <span className={className}>{displayText}</span>;
   752→   752→}
   753→   753→```

   820→   820→  containerSelector: string,
   821→   821→  itemsSelector: string
   822→   822→) {
   823→   823→  const container = document.querySelector(containerSelector);
   824→   824→  const items = document.querySelectorAll(itemsSelector);
   825→   825→

   838→   838→      scrub: 1,
   839→   839→      pin: true,
   840→   840→      anticipatePin: 1,
   841→   841→    },
   842→   842→  });

  1018→  1018→
  1019→  1019→1. **GPU-accelerated properties only**: `transform`, `opacity`
  1020→  1020→2. **Use `will-change` sparingly** — only on actively animating elements

  1022→  1022→4. **Lazy mount** — dynamic import heavy animation components
  1023→  1023→5. **Debounce scroll handlers** — use `requestAnimationFrame`
